--!nocheck
-- ============================================================
-- RARE INDICATOR  —  ESP + AUTO-AIM (Delta / executor ready)
-- Highlights every "Rare Candidate" (Model-wrapped block slots in
-- workspace.Blocks) and silently aims the camera at the nearest one.
-- Two toggles. No walking. No mining.
-- Hotkey to show/hide GUI: RightShift
-- ============================================================

local Players          = game:GetService("Players")
local RunService       = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local TweenService     = game:GetService("TweenService")
local Workspace        = game:GetService("Workspace")

local LocalPlayer = Players.LocalPlayer
local PlayerGui   = LocalPlayer:WaitForChild("PlayerGui")
local Camera      = Workspace.CurrentCamera

-- ============================================================
-- CONFIG
-- ============================================================
local TOGGLE_KEY    = Enum.KeyCode.RightShift
local AIM_RANGE     = 500          -- studs; nearest within this radius is aimed at
local CAMERA_LERP   = 0.25         -- 0..1; lower = smoother, higher = snappier
local ESP_COLOR     = Color3.fromRGB(245, 90, 165)   -- pink
local ESP_OUTLINE   = Color3.fromRGB(255, 255, 255)
local SCAN_INTERVAL = 0.25         -- seconds between refresh sweeps

-- ============================================================
-- STATE
-- ============================================================
local state = {
	espOn = false,
	aimOn = false,
}

local blocksFolder = Workspace:FindFirstChild("Blocks")
local tracked = {}   -- [model] = Highlight
local addedConn, removedConn, foundConn

-- ============================================================
-- RARE CANDIDATE DETECTION
-- A "rare candidate" = a Model living inside workspace.Blocks
-- whose direct or nested parts make up an ore slot.
-- ============================================================
local function isRareCandidateModel(inst)
	if not inst or not inst:IsA("Model") then return false end
	if not blocksFolder then return false end
	if not inst:IsDescendantOf(blocksFolder) then return false end
	-- Must contain at least one BasePart
	for _, d in ipairs(inst:GetDescendants()) do
		if d:IsA("BasePart") then return true end
	end
	return false
end

local function modelPos(model)
	local ok, cf = pcall(function() return model:GetPivot() end)
	if ok and cf then return cf.Position end
	return nil
end

-- ============================================================
-- ESP HIGHLIGHTS
-- ============================================================
local function addHighlight(model)
	if tracked[model] then return end
	local h = Instance.new("Highlight")
	h.Name = "_RareESP"
	h.FillColor = ESP_COLOR
	h.OutlineColor = ESP_OUTLINE
	h.FillTransparency = 0.55
	h.OutlineTransparency = 0
	h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	h.Adornee = model
	h.Parent = model
	tracked[model] = h
end

local function removeHighlight(model)
	local h = tracked[model]
	if h then h:Destroy() end
	tracked[model] = nil
end

local function clearAll()
	for m, h in pairs(tracked) do
		if h then h:Destroy() end
		tracked[m] = nil
	end
end

local function fullScan()
	clearAll()
	if not state.espOn then return end
	if not blocksFolder then return end
	for _, m in ipairs(blocksFolder:GetDescendants()) do
		if m:IsA("Model") and isRareCandidateModel(m) then
			addHighlight(m)
		end
	end
end

-- ============================================================
-- LISTEN FOR NEW/REMOVED CANDIDATES
-- ============================================================
local function attachBlocksListeners()
	if addedConn then addedConn:Disconnect() end
	if removedConn then removedConn:Disconnect() end
	if not blocksFolder then return end

	addedConn = blocksFolder.DescendantAdded:Connect(function(d)
		if not state.espOn then return end
		if d:IsA("Model") and isRareCandidateModel(d) then
			addHighlight(d)
		elseif d:IsA("BasePart") then
			-- A part just appeared; its parent model may now qualify
			local p = d.Parent
			if p and p:IsA("Model") and isRareCandidateModel(p) then
				addHighlight(p)
			end
		end
	end)

	removedConn = blocksFolder.DescendantRemoving:Connect(function(d)
		if d:IsA("Model") and tracked[d] then
			removeHighlight(d)
		end
	end)
end

local function bindBlocksFolder()
	blocksFolder = Workspace:FindFirstChild("Blocks")
	attachBlocksListeners()

	if foundConn then foundConn:Disconnect() end
	foundConn = Workspace.ChildAdded:Connect(function(c)
		if c.Name == "Blocks" and c:IsA("Folder") then
			blocksFolder = c
			attachBlocksListeners()
			if state.espOn then fullScan() end
		end
	end)
	Workspace.ChildRemoved:Connect(function(c)
		if c == blocksFolder then
			blocksFolder = nil
			clearAll()
		end
	end)
end

-- Periodic safety re-scan (catches anything the events miss)
task.spawn(function()
	while true do
		task.wait(SCAN_INTERVAL)
		if state.espOn and blocksFolder then
			for _, m in ipairs(blocksFolder:GetDescendants()) do
				if m:IsA("Model") and not tracked[m] and isRareCandidateModel(m) then
					addHighlight(m)
				end
			end
			-- prune dead refs
			for m in pairs(tracked) do
				if not m.Parent then removeHighlight(m) end
			end
		end
	end
end)

-- ============================================================
-- AUTO-AIM (camera only — no walking, no jumping, no mining)
-- ============================================================
local function findNearestRare(origin)
	if not blocksFolder then return nil, math.huge end
	local best, bd = nil, math.huge
	-- Prefer the tracked set if ESP is on (cheaper); else walk descendants
	if next(tracked) then
		for m in pairs(tracked) do
			if m.Parent then
				local p = modelPos(m)
				if p then
					local d = (p - origin).Magnitude
					if d < bd and d <= AIM_RANGE then best, bd = m, d end
				end
			end
		end
	else
		for _, m in ipairs(blocksFolder:GetDescendants()) do
			if m:IsA("Model") and isRareCandidateModel(m) then
				local p = modelPos(m)
				if p then
					local d = (p - origin).Magnitude
					if d < bd and d <= AIM_RANGE then best, bd = m, d end
				end
			end
		end
	end
	return best, bd
end

RunService.RenderStepped:Connect(function()
	if not state.aimOn then return end
	local char = LocalPlayer.Character
	local hrp  = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	local target = findNearestRare(hrp.Position)
	if not target then return end
	local tp = modelPos(target)
	if not tp then return end

	Camera.CFrame = Camera.CFrame:Lerp(
		CFrame.lookAt(Camera.CFrame.Position, tp),
		CAMERA_LERP
	)
end)

local C = {
	BG    = Color3.fromRGB(14, 19, 33),
	DEEP  = Color3.fromRGB(5, 8, 16),

	HOVER = Color3.fromRGB(32, 44, 70),

local function corner(o, r) local c = Instance.new("UICorner") c.CornerRadius = UDim.new(0, r or 6) c.Parent = o return c end
local function stroke(o, col, t) local s = Instance.new("UIStroke") s.Color = col s.Thickness = t or 1 s.Parent = o return s end

local function tween(o, props, t)
	return TweenService:Create(o, TweenInfo.new(t or 0.18, Enum.EasingStyle.Quart), props)
end

local function hoverFX(btn, normal, hover)
	btn.MouseEnter:Connect(function() tween(btn, {BackgroundColor3 = hover}):Play() end)
	btn.MouseLeave:Connect(function() tween(btn, {BackgroundColor3 = normal}):Play() end)
end

local function makeToggle(initial, onChange)
	local container = Instance.new("Frame")
	container.Size = UDim2.new(0, 44, 0, 22)
	container.BackgroundColor3 = initial and C.ACC or C.DEEP
	container.BorderSizePixel = 0
	corner(container, 11)

	local knob = Instance.new("Frame")
	knob.Size = UDim2.new(0, 16, 0, 16)
	knob.Position = initial and UDim2.new(1, -19, 0.5, -8) or UDim2.new(0, 3, 0.5, -8)
	knob.BackgroundColor3 = C.TXT
	knob.BorderSizePixel = 0
	knob.Parent = container
	corner(knob, 8)

	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, 0, 1, 0)
	btn.BackgroundTransparency = 1
	btn.Text = ""
	btn.Parent = container

	local cur = initial
	local function set(v, fire)
		cur = v
		tween(knob, {Position = v and UDim2.new(1, -19, 0.5, -8) or UDim2.new(0, 3, 0.5, -8)}):Play()
		tween(container, {BackgroundColor3 = v and C.ACC or C.DEEP}):Play()
		if fire and onChange then onChange(v) end
	end
	btn.MouseButton1Click:Connect(function() set(not cur, true) end)
	return container, set
end

	while true do
		task.wait(0.4)
		if espSub.Parent then
			local n = 0
			for _ in pairs(tracked) do n = n + 1 end
			espSub.Text = state.espOn and (n .. " indicator" .. (n == 1 and "" or "s")) or "off"
		end
		if aimSub.Parent then
			aimSub.Text = state.aimOn and ("aiming · " .. AIM_RANGE .. "st range") or "off"
		end

-- Drag
do
	local dragging, ds, sp = false, nil, nil
	title.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true; ds = input.Position; sp = root.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local d = input.Position - ds
		root.Position = UDim2.new(sp.X.Scale, sp.X.Offset + d.X,
			sp.Y.Scale, sp.Y.Offset + d.Y)
	end)
end

local function show() root.Visible = true;  bubble.Visible = false end
local function hide() root.Visible = false; bubble.Visible = true  end

closeBtn.MouseButton1Click:Connect(hide)

do
	local DRAG_THRESH = 6
	local dragging, moved, ds, sp = false, false, nil, nil
	bubble.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true; moved = false; ds = input.Position; sp = bubble.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					if not moved then show() end
				end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local d = input.Position - ds
		if not moved and (math.abs(d.X) > DRAG_THRESH or math.abs(d.Y) > DRAG_THRESH) then
			moved = true
		end
		if moved then
			bubble.Position = UDim2.new(sp.X.Scale, sp.X.Offset + d.X,
				sp.Y.Scale, sp.Y.Offset + d.Y)
		end
	end)
end

-- Toggle key
UserInputService.InputBegan:Connect(function(input, gpe)
	if gpe then return end
	if input.KeyCode == TOGGLE_KEY then
		if root.Visible then hide() else show() end
	end
end)
