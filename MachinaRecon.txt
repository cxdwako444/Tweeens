local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end

-- ============================================================
-- IN-MEMORY LOG (no print — anti-cheat reads it)

local LOG_LINES = {}
local LOG_MAX = 250
local function log(s)
	table.insert(LOG_LINES, tostring(s))
	if #LOG_LINES > LOG_MAX then table.remove(LOG_LINES, 1) end
end

-- ============================================================
-- BUFFER BUILDER (the breakthrough)
-- ============================================================
local function makeMineBuffer(x, y, z, dmg)
	local b = buffer.create(32)
	buffer.writef64(b, 0,  x)
	buffer.writef64(b, 8,  y)
	buffer.writef64(b, 16, z)
	buffer.writef64(b, 24, dmg)
	return b
end

local function randomNonce()
	return string.char(
		math.random(0,255), math.random(0,255),
		math.random(0,255), math.random(0,255))
end

local function fireMine(pos, dmg)
	pcall(function()
		R:FireServer(randomNonce(), "DamageBlockSmooth",
			makeMineBuffer(pos.X, pos.Y, pos.Z, dmg))
	end)
end

-- ============================================================
-- HEX (for captures, kept from v5)
-- ============================================================
local function strHex(s, max)
	max = max or 64
	local n = math.min(#s, max)
	local out = {}
	for i = 1, n do out[i] = string.format("%02x", string.byte(s, i)) end
	local h = table.concat(out, " ")
	if #s > n then h = h .. (" ...+%d"):format(#s - n) end
	return h
end
local function bufHex(b, max)
	local ok, raw = pcall(buffer.tostring, b)
	if ok and type(raw) == "string" then return strHex(raw, max or 200), #raw end
	return "<unreadable>", 0
end

local stats = {
	totalFires = 0, mineFires = 0,
	mineCaptureRemaining = 0,
	hookEnabled = true,
	radiusOn = false,
	radius = 8,
	rateHz = 5,
	damage = 0.05,
	radiusBlocksLastCycle = 0,
}
local CAPTURED = {}

-- ============================================================
-- NAMECALL HOOK (counts only)
-- ============================================================
pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		if stats.hookEnabled then
			local method = getnamecallmethod and getnamecallmethod() or ""
			if method == "FireServer" or method == "InvokeServer" then
				stats.totalFires = stats.totalFires + 1
				local a2 = select(2, ...)
				if type(a2) == "string" and a2 == "DamageBlockSmooth" then
					stats.mineFires = stats.mineFires + 1
					if stats.mineCaptureRemaining > 0 then
						stats.mineCaptureRemaining = stats.mineCaptureRemaining - 1
						local a1 = select(1, ...)
						local a3 = select(3, ...)
						local hex, len = "?", 0
						if typeof(a3) == "buffer" then hex, len = bufHex(a3, 256) end
						local nhex = type(a1) == "string" and strHex(a1, 8) or "?"
						table.insert(CAPTURED, {
							idx = #CAPTURED + 1, hex = hex, len = len,
							nonce = nhex, fireNo = stats.totalFires,
						})
					end
				end
			end
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)

-- ============================================================
-- RADIUS MINE LOOP
-- ============================================================
local function getNearbyBlocks(origin, radius)
	local blocksFolder = workspace:FindFirstChild("Blocks")
	if not blocksFolder then return {} end
	local result = {}
	for _, d in ipairs(blocksFolder:GetDescendants()) do
		if d:IsA("BasePart") then
			local p = d.Position
			if (p - origin).Magnitude <= radius then
				table.insert(result, p)
			end
		end
	end
	return result
end

task.spawn(function()
	while true do
		if stats.radiusOn then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local positions = getNearbyBlocks(hrp.Position, stats.radius)
				stats.radiusBlocksLastCycle = #positions
				for _, p in ipairs(positions) do
					fireMine(p, stats.damage)
				end
			else
				stats.radiusBlocksLastCycle = 0
			end
			task.wait(1 / math.max(1, stats.rateHz))
		else
			task.wait(0.1)
		end
	end
end)

btn(140, "🔨 Test 1 Block (nearest)", C.GOLD, function()
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then log("test: no character"); return end
	local nearest, nd = nil, math.huge
	local blocks = workspace:FindFirstChild("Blocks")
	if not blocks then log("test: no workspace.Blocks"); return end
	for _, d in ipairs(blocks:GetDescendants()) do
		if d:IsA("BasePart") then
			local dist = (d.Position - hrp.Position).Magnitude
			if dist < nd then nearest, nd = d, dist end
		end
	end
	if not nearest then log("test: no blocks"); return end
	local p = nearest.Position
	log(("test: mining nearest at (%.1f,%.1f,%.1f) d=%.1f dmg=1"):format(p.X, p.Y, p.Z, nd))
	-- Send one big damage hit
	fireMine(p, 1.0)
end)

	lines[#lines+1] = ""

		lines[#lines+1] = ("[#%d fire=%d] nonce=%s len=%d"):format(c.idx, c.fireNo, c.nonce, c.len)

	fileBtn.Text = ("✓ %d bytes"):format(#payload)
	task.delay(2.5, function() if fileBtn.Parent then fileBtn.Text = "💾 Save Log to File" end end)
end)

	end)
