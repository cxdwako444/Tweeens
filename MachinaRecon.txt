-- ENV PROBE (silent)

local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end

-- ============================================================
-- IN-MEMORY LOG (no print — anti-cheat reads print output)
-- ============================================================
local LOG_LINES = {}
local LOG_MAX = 300
local function log(s)
	table.insert(LOG_LINES, tostring(s))
	if #LOG_LINES > LOG_MAX then table.remove(LOG_LINES, 1) end
end

-- ============================================================
-- HEX
-- ============================================================
local function strHex(s, max)
	max = max or 64
	local n = math.min(#s, max)
	local out = {}
	for i = 1, n do out[i] = string.format("%02x", string.byte(s, i)) end
	local h = table.concat(out, " ")
	if #s > n then h = h .. (" ...+%d"):format(#s - n) end
	return h
end
local function bufRaw(b)
	local ok, raw = pcall(buffer.tostring, b)
	if ok and type(raw) == "string" then return raw end
	return nil
end
local function bufHex(b, max)
	local raw = bufRaw(b)
	if not raw then return "<unreadable>", 0 end
	return strHex(raw, max or 200), #raw
end

-- ============================================================
-- STATE
-- ============================================================
local stats = {
	totalFires = 0,
	mineFires = 0,
	mineCaptureRemaining = 0,
	markCount = 0,
	hookEnabled = true,
}
local CAPTURED_MINES = {}  -- {hex, len, nonceHex, time}

-- ============================================================
-- HOOK — counts only by default
-- ============================================================
pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		if stats.hookEnabled then
			local method = getnamecallmethod and getnamecallmethod() or ""
			if method == "FireServer" or method == "InvokeServer" then
				stats.totalFires = stats.totalFires + 1
				local a1 = select(1, ...)
				local a2 = select(2, ...)
				if type(a2) == "string" and a2 == "DamageBlockSmooth" then
					stats.mineFires = stats.mineFires + 1
					if stats.mineCaptureRemaining > 0 then
						stats.mineCaptureRemaining = stats.mineCaptureRemaining - 1
						local a3 = select(3, ...)
						local hex, len = "?", 0
						if typeof(a3) == "buffer" then hex, len = bufHex(a3, 256) end
						local nhex = type(a1) == "string" and strHex(a1, 8) or "?"
						table.insert(CAPTURED_MINES, {
							idx = #CAPTURED_MINES + 1,
							hex = hex,
							len = len,
							nonce = nhex,
							t = tick(),
							fireNo = stats.totalFires,
						})
					end
				end
			end
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)

local fileBtn
fileBtn = btn(206, "💾 Save Log to File", C.BLUE, function()
	if not _writefile then fileBtn.Text="✗ no writefile"; return end
	-- Build a full report from in-memory log + captures
	local lines = {}
	for _, l in ipairs(LOG_LINES) do lines[#lines+1] = l end
	lines[#lines+1] = ""
	lines[#lines+1] = "===== CAPTURED MINE BUFFERS ====="
	for _, c in ipairs(CAPTURED_MINES) do
		lines[#lines+1] = ("[#%d fire=%d] nonce=%s len=%d"):format(
			c.idx, c.fireNo, c.nonce, c.len)
		lines[#lines+1] = "  hex: " .. c.hex
	end
	local payload = table.concat(lines, "\n")
	local ok, err = pcall(_writefile, OUT_FILE, payload)
	if not ok then
		log("ERR writefile: " .. tostring(err))
		fileBtn.Text = "✗ FAIL"; return
	end
	fileBtn.Text = ("✓ saved %d bytes"):format(#payload)
	task.delay(2.5, function() if fileBtn.Parent then fileBtn.Text = "💾 Save Log to File" end end)
end)

do
	local dragging, ds, sp_ = false, nil, nil
	titleBar.InputBegan:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1
			or input.UserInputType==Enum.UserInputType.Touch then
			dragging=true; ds=input.Position; sp_=root.Position
			input.Changed:Connect(function()
				if input.UserInputState==Enum.UserInputState.End then dragging=false end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType~=Enum.UserInputType.MouseMovement
			and input.UserInputType~=Enum.UserInputType.Touch then return end
		local d = input.Position - ds
		root.Position = UDim2.new(sp_.X.Scale, sp_.X.Offset+d.X,
			sp_.Y.Scale, sp_.Y.Offset+d.Y)
	end)
end

				for _, c in ipairs(CAPTURED_MINES) do
