local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end

-- ============================================================
-- LOG (memory only)
-- ============================================================
local LOG = {}
local LOG_MAX = 400
local function log(s)
	table.insert(LOG, tostring(s))
	if #LOG > LOG_MAX then table.remove(LOG, 1) end
end

-- ============================================================
-- HEX
-- ============================================================
local function strHex(s, max)
	max = max or 64; local n = math.min(#s, max); local out = {}
	for i = 1, n do out[i] = string.format("%02x", string.byte(s, i)) end
	local h = table.concat(out, " ")
	if #s > n then h = h .. (" +%d"):format(#s - n) end
	return h
end
local function bufHex(b, max)
	local ok, raw = pcall(buffer.tostring, b)
	if ok and type(raw) == "string" then return strHex(raw, max or 200), #raw end
	return "<no>", 0
end
local function decode4f64(b)
	local ok, raw = pcall(buffer.tostring, b)
	if not ok or #raw < 32 then return nil end
	local v = {}
	for i = 0, 3 do
		v[i+1] = buffer.readf64(b, i * 8)
	end
	return v
end

-- ============================================================
-- BLOCK INSPECTOR (the key feature)
-- ============================================================
local function partInfo(p, prefix)
	prefix = prefix or "  "
	log(prefix .. ("[%s] %s"):format(p.ClassName, p:GetFullName()))
	if p:IsA("BasePart") then
		local pos = p.Position
		log(prefix .. ("  Position=(%.1f, %.1f, %.1f)"):format(pos.X, pos.Y, pos.Z))
		log(prefix .. ("  Size=(%.1f, %.1f, %.1f)"):format(p.Size.X, p.Size.Y, p.Size.Z))
		log(prefix .. ("  Material=%s  MV=%s"):format(tostring(p.Material), tostring(p.MaterialVariant)))
	end
	local attrs = p:GetAttributes()
	local hasAttrs = false
	for k, v in pairs(attrs) do
		hasAttrs = true
		log(prefix .. ("  @%s = %s (%s)"):format(k, tostring(v), typeof(v)))
	end
	if not hasAttrs then log(prefix .. "  (no attributes)") end
	-- Children
	local kids = p:GetChildren()
	if #kids > 0 then
		log(prefix .. ("  %d children:"):format(#kids))
		for i, c in ipairs(kids) do
			if i > 6 then log(prefix .. "    ...more"); break end
			log(prefix .. ("    %s [%s]"):format(c.Name, c.ClassName))
		end
	end
end

local function inspectSampleBlocks()
	local blocks = Workspace:FindFirstChild("Blocks")
	if not blocks then log("no workspace.Blocks"); return end
	log("=== INSPECT 3 SAMPLE BLOCKS ===")
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	-- Get nearest 3 BaseParts
	local arr = {}
	for _, d in ipairs(blocks:GetDescendants()) do
		if d:IsA("BasePart") then
			local dist = hrp and (d.Position - hrp.Position).Magnitude or 0
			table.insert(arr, {part=d, dist=dist})
		end
	end
	table.sort(arr, function(a, b) return a.dist < b.dist end)
	for i = 1, math.min(3, #arr) do
		log(("--- Block #%d (dist %.1f) ---"):format(i, arr[i].dist))
		partInfo(arr[i].part)
		-- Also inspect parent chain up to Blocks folder
		local cur = arr[i].part.Parent
		local depth = 0
		while cur and cur ~= Workspace and depth < 4 do
			log("  ↑ Parent: " .. cur:GetFullName())
			local pattrs = cur:GetAttributes()
			for k, v in pairs(pattrs) do
				log(("    @%s = %s (%s)"):format(k, tostring(v), typeof(v)))
			end
			cur = cur.Parent; depth = depth + 1
		end
	end
	log(("(workspace.Blocks has %d total BaseParts)"):format(#arr))
end

local stats = {
	totalFires = 0, mineFires = 0,
	captureRemaining = 0,
	hookEnabled = true,
	markCount = 0,
}

-- ============================================================
-- HOOK
-- ============================================================
pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		if stats.hookEnabled then
			local method = getnamecallmethod and getnamecallmethod() or ""
			if method == "FireServer" or method == "InvokeServer" then
				stats.totalFires = stats.totalFires + 1
				local a2 = select(2, ...)
				if type(a2) == "string" and a2 == "DamageBlockSmooth" then
					stats.mineFires = stats.mineFires + 1
					if stats.captureRemaining > 0 then
						stats.captureRemaining = stats.captureRemaining - 1
						local a1 = select(1, ...)
						local a3 = select(3, ...)
						local nhex = type(a1) == "string" and strHex(a1, 8) or "?"
						local hex, len = "?", 0
						local decoded = nil
						if typeof(a3) == "buffer" then
							hex, len = bufHex(a3, 256)
							decoded = decode4f64(a3)
						end
						log(("[CAP %d] nonce=%s len=%d"):format(stats.mineFires, nhex, len))
						log("  hex: " .. hex)
						if decoded then
							log(("  decoded: X=%.3f Y=%.3f Z=%.3f dmg=%.4f"):format(
								decoded[1], decoded[2], decoded[3], decoded[4]))
						end
						-- AUTO-INSPECT: raycast from camera to find what part was hit
						local cam = Workspace.CurrentCamera
						if cam then
							local rp = RaycastParams.new()
							rp.FilterType = Enum.RaycastFilterType.Exclude
							local ch = LocalPlayer.Character
							if ch then rp.FilterDescendantsInstances = {ch} end
							local origin = cam.CFrame.Position
							local dir = cam.CFrame.LookVector * 80
							local hit = Workspace:Raycast(origin, dir, rp)
							if hit and hit.Instance then
								log("  → hit part:")
								partInfo(hit.Instance, "    ")
							else
								log("  → raycast missed")
							end
						end
					end
				end
			end
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)
