-- FS (Delta: writefile / appendfile / readfile / isfile)

local function get(n) return rawget(getfenv(0), n) or rawget(_G, n) end
local _writefile  = writefile  or get("writefile")
local _appendfile = appendfile or get("appendfile")
local _readfile   = readfile   or get("readfile")
local _isfile     = isfile     or get("isfile")

local _decompile  = decompile  or get("decompile")
local _getscripts = getscripts or get("getscripts") or get("get_scripts")
local _getloaded  = getloadedmodules or get("getloadedmodules") or get("get_loaded_modules")
local _getbyte    = getscriptbytecode or get("getscriptbytecode")

if not _writefile then
	warn("[Recon] No writefile() — Delta needed.")
end

local _buf, _saving = {}, false
local function flush()
	if _saving or #_buf == 0 or not _writefile then return end
	_saving = true
	local chunk = table.concat(_buf, "\n") .. "\n"
	_buf = {}
	pcall(function()
		if _appendfile and _isfile and _isfile(OUT_FILE) then
			_appendfile(OUT_FILE, chunk)
		else
			local prev = ""
			if _readfile and _isfile and _isfile(OUT_FILE) then
				prev = _readfile(OUT_FILE) or ""
			end
			_writefile(OUT_FILE, prev .. chunk)
		end
	end)
	_saving = false
end

local function log(line)
	table.insert(_buf, line)
	if #_buf >= 20 then flush() end
end

-- wipe file on each run
if _writefile then pcall(function() _writefile(OUT_FILE, "") end) end

task.spawn(function()
	while true do task.wait(1.5); flush() end
end)

-- ============================================================
-- HELPERS
-- ============================================================
local function flag(s)
	if type(s) ~= "string" then return false end
	local low = s:lower()
	for _, k in ipairs(KEYWORDS) do
		if low:find(k, 1, true) then return true end
	end
	return false
end

local function banner(title)
	log("")
	log("################################################################")
	log("###  " .. title)
	log("################################################################")
end

local function section(title)
	log("")
	log("----- " .. title .. " -----")
end

local function tryDecompile(s)
	if not _decompile then return nil, "no-decompile" end
	local done, src, err = false, nil, nil
	task.spawn(function()
		local ok, res = pcall(_decompile, s)
		if ok then src = res else err = tostring(res) end
		done = true
	end)
	local t0 = tick()
	while not done and (tick() - t0) < DECOMPILE_TIME do task.wait(0.1) end
	if not done then return nil, "timeout" end
	if not src or src == "" then return nil, err or "empty" end
	if #src > MAX_SRC_BYTES then
		src = src:sub(1, MAX_SRC_BYTES) .. "\n-- [TRUNCATED at " .. MAX_SRC_BYTES .. " bytes] --"
	end
	return src
end

local function dumpScriptSource(s, indent)
	indent = indent or ""
	local src, err = tryDecompile(s)
	if not src then
		log(indent .. "  -- <decompile failed: " .. tostring(err) .. ">")
		-- fallback: try bytecode (rare but useful)
		if _getbyte then
			local ok, bc = pcall(_getbyte, s)
			if ok and bc and #bc > 0 then
				log(indent .. "  -- bytecode size: " .. #bc .. " bytes")
			end
		end
		return
	end
	log(indent .. "  >>>>>>> SOURCE >>>>>>>")
	for line in src:gmatch("([^\n]*)\n?") do
		log(indent .. line)
	end
	log(indent .. "  <<<<<<< END SOURCE <<<<<<<")
end

-- ============================================================
-- SERIALIZER for remote args
-- ============================================================
local function ser(v, d)
	d = d or 0
	if d > 3 then return "<deep>" end
	local t = typeof(v)
	if t == "Instance" then
		local cls, fn = "?", "?"
		pcall(function() cls = v.ClassName end)
		pcall(function() fn  = v:GetFullName() end)
		return ("<%s:%s>"):format(cls, fn)
	elseif t == "Vector3" then return ("V3(%.1f,%.1f,%.1f)"):format(v.X, v.Y, v.Z)
	elseif t == "Vector2" then return ("V2(%.1f,%.1f)"):format(v.X, v.Y)
	elseif t == "CFrame"  then local p = v.Position; return ("CF(%.1f,%.1f,%.1f)"):format(p.X, p.Y, p.Z)
	elseif t == "Color3"  then return ("C3(%d,%d,%d)"):format(v.R*255, v.G*255, v.B*255)
	elseif t == "EnumItem" then return tostring(v)
	elseif t == "string" then return ("%q"):format(v)
	elseif t == "number" or t == "boolean" or t == "nil" then return tostring(v)
	elseif t == "table" then
		local parts, n = {}, 0
		for k, vv in pairs(v) do
			n = n + 1
			if n > 16 then table.insert(parts, "..."); break end
			table.insert(parts, ("[%s]=%s"):format(ser(k, d+1), ser(vv, d+1)))
		end
		return "{" .. table.concat(parts, ", ") .. "}"
	end
	return ("<%s>"):format(t)
end

local function serArgs(...)
	local n, parts = select("#", ...), {}
	for i = 1, n do parts[i] = ser((select(i, ...))) end
	return table.concat(parts, " | ")
end

local function dumpRemotes(parent, label)
	section("Remotes in " .. label)
	local count, flagged = 0, 0
	for _, d in ipairs(parent:GetDescendants()) do
		if d:IsA("RemoteEvent") or d:IsA("RemoteFunction")
			or d:IsA("BindableEvent") or d:IsA("BindableFunction")
			or d:IsA("UnreliableRemoteEvent") then
			count = count + 1
			local mark = flag(d.Name) and "  *** FLAGGED ***" or ""
			if mark ~= "" then flagged = flagged + 1 end
			log(("[%s] %s%s"):format(d.ClassName, d:GetFullName(), mark))
		end
	end
	log(("(total %d, flagged %d)"):format(count, flagged))
end

local function dumpTool(t, where)
	section(("Tool [%s]  %s"):format(where, t.Name))
	log("  Path: " .. t:GetFullName())
	log("  Attributes:")
	for k, v in pairs(t:GetAttributes()) do
		log(("    @%s = %s"):format(k, ser(v)))
	end
	-- list everything inside
	for _, child in ipairs(t:GetDescendants()) do
		local cls = child.ClassName
		if child:IsA("LocalScript") or child:IsA("ModuleScript") or child:IsA("Script") then
			log(("  -- script: [%s] %s"):format(cls, child:GetFullName()))
			dumpScriptSource(child, "    ")
		elseif child:IsA("RemoteEvent") or child:IsA("RemoteFunction")
			or child:IsA("BindableEvent") or child:IsA("BindableFunction") then
			log(("  -- remote: [%s] %s"):format(cls, child:GetFullName()))
		elseif child:IsA("ValueBase") then
			local ok, val = pcall(function() return child.Value end)
			log(("  -- value:  [%s] %s = %s"):format(cls, child.Name, ok and ser(val) or "?"))
		elseif child:IsA("BasePart") or child:IsA("Folder") or child:IsA("Configuration") then
			log(("  -- %s: %s"):format(cls, child.Name))
		end
	end
end

local function dumpToolsIn(parent, label)
	if not parent then return end
	for _, c in ipairs(parent:GetChildren()) do
		if c:IsA("Tool") then dumpTool(c, label) end
	end
end
dumpToolsIn(LocalPlayer:FindFirstChild("Backpack"), "Backpack")

local seenSrc = {}

local function dumpModuleIfRelevant(m, why)
	if seenSrc[m] then return end
	seenSrc[m] = true
	if not (m:IsA("ModuleScript") or m:IsA("LocalScript") or m:IsA("Script")) then return end
	local path = m:GetFullName()
	if not (flag(m.Name) or flag(path)) then return end
	section(("[%s]  %s   (matched: %s)"):format(m.ClassName, path, why))
	dumpScriptSource(m, "  ")
end

-- ReplicatedStorage descendants

	dumpModuleIfRelevant(d, "ReplicatedStorage")
end

	for _, d in ipairs(StarterPlayer:GetDescendants()) do
		dumpModuleIfRelevant(d, "StarterPlayer")

local fireCount, markCount = 0, 0

local function appendFireLine(self, method, ...)
	fireCount = fireCount + 1
	pcall(function()
		log(("[FIRE #%d] %s :%s(%s)"):format(
			fireCount, self:GetFullName(), method, serArgs(...)))
	end)
end

local function installHook()
	local ok, mt = pcall(getrawmetatable, game)
	if not ok or not mt then
		log("!! getrawmetatable failed — no live log available")
		return false
	end
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end

	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		local method = getnamecallmethod and getnamecallmethod() or ""
		if method == "FireServer" or method == "InvokeServer"
			or method == "fireServer" or method == "invokeServer" then
			appendFireLine(self, method, ...)
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
	return true
end

if installHook() then
	log("[Recon] Namecall hook installed.")
else
	log("[Recon] Could not install namecall hook.")
end
