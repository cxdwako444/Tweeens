-- FILESYSTEM (Delta exposes writefile/appendfile/isfile)
-- ============================================================
local OUT_FILE = "MachinaRecon.txt"

local _writefile  = rawget(getfenv(), "writefile")  or writefile
local _appendfile = rawget(getfenv(), "appendfile") or appendfile
local _readfile   = rawget(getfenv(), "readfile")   or readfile
local _isfile     = rawget(getfenv(), "isfile")     or isfile

if not _writefile then
	warn("[Recon] No writefile() — this executor cannot save files.")
end

local _buffer = {}
local _saving = false
local function flush()
	if _saving or #_buffer == 0 then return end
	_saving = true
	local chunk = table.concat(_buffer, "\n") .. "\n"
	_buffer = {}
	pcall(function()
		if _appendfile and _isfile and _isfile(OUT_FILE) then
			_appendfile(OUT_FILE, chunk)
		elseif _writefile then
			local prev = ""
			if _readfile and _isfile and _isfile(OUT_FILE) then
				prev = _readfile(OUT_FILE) or ""
			end
			_writefile(OUT_FILE, prev .. chunk)
		end
	end)
	_saving = false
end

local function log(line)
	table.insert(_buffer, line)
	if #_buffer >= 8 then flush() end
end

-- Wipe previous log on each run so it doesn't grow forever
if _writefile then pcall(function() _writefile(OUT_FILE, "") end) end

-- Periodic flush
task.spawn(function()
	while true do
		task.wait(1.5)
		flush()
	end
end)

-- ============================================================
-- ARG SERIALIZER  (handles Instance / Vector3 / CFrame / tables)
-- ============================================================
local function ser(v, depth)
	depth = depth or 0
	if depth > 3 then return "<deep>" end
	local t = typeof(v)
	if t == "Instance" then
		local cls, fn = "?", "?"
		pcall(function() cls = v.ClassName end)
		pcall(function() fn  = v:GetFullName() end)
		return ("<%s:%s>"):format(cls, fn)
	elseif t == "Vector3" then
		return ("V3(%.1f,%.1f,%.1f)"):format(v.X, v.Y, v.Z)
	elseif t == "Vector2" then
		return ("V2(%.1f,%.1f)"):format(v.X, v.Y)
	elseif t == "CFrame" then
		local p = v.Position
		return ("CF(%.1f,%.1f,%.1f)"):format(p.X, p.Y, p.Z)
	elseif t == "Color3" then
		return ("C3(%d,%d,%d)"):format(v.R*255, v.G*255, v.B*255)
	elseif t == "EnumItem" then
		return tostring(v)
	elseif t == "string" then
		return ("%q"):format(v)
	elseif t == "number" or t == "boolean" or t == "nil" then
		return tostring(v)
	elseif t == "table" then
		local parts = {}
		local count = 0
		for k, vv in pairs(v) do
			count = count + 1
			if count > 16 then table.insert(parts, "...") break end
			table.insert(parts, ("[%s]=%s"):format(ser(k, depth+1), ser(vv, depth+1)))
		end
		return "{" .. table.concat(parts, ", ") .. "}"
	end
	return ("<%s>"):format(t)
end

local function serArgs(...)
	local n = select("#", ...)
	local parts = {}
	for i = 1, n do
		parts[i] = ser((select(i, ...)))
	end
	return table.concat(parts, " | ")
end

local KEYWORDS = {
	"mine", "break", "hit", "auto", "machina", "relic",
	"ore", "block", "dig", "pickaxe", "swing", "place",
	"sell", "inventory", "use", "activate", "tool",
}

local function flag(name)
	local low = name:lower()
	for _, k in ipairs(KEYWORDS) do
		if low:find(k, 1, true) then return true end
	end
	return false
end

local function dumpRemotesIn(parent, label)
	log(("----- REMOTES in %s -----"):format(label))
	local count, flagged = 0, 0
	for _, d in ipairs(parent:GetDescendants()) do
		if d:IsA("RemoteEvent") or d:IsA("RemoteFunction")
			or d:IsA("BindableEvent") or d:IsA("BindableFunction")
			or d:IsA("UnreliableRemoteEvent") then
			count = count + 1
			local mark = flag(d.Name) and "  *** FLAGGED ***" or ""
			if mark ~= "" then flagged = flagged + 1 end
			log(("[%s] %s%s"):format(d.ClassName, d:GetFullName(), mark))
		end
	end
	log(("(total %d, flagged %d)"):format(count, flagged))
	log("")
end

dumpRemotesIn(ReplicatedStorage, "ReplicatedStorage")
pcall(function() dumpRemotesIn(game:GetService("ReplicatedFirst"), "ReplicatedFirst") end)
pcall(function() dumpRemotesIn(workspace, "Workspace") end)

local function dumpTools(parent, label)
	for _, t in ipairs(parent:GetChildren()) do
		if t:IsA("Tool") then
			log(("[Tool@%s] %s"):format(label, t.Name))
			for _, child in ipairs(t:GetDescendants()) do
				if child:IsA("LocalScript") or child:IsA("ModuleScript") or child:IsA("Script") then
					log(("    └ %s  %s"):format(child.ClassName, child:GetFullName()))
				elseif child:IsA("RemoteEvent") or child:IsA("RemoteFunction")
					or child:IsA("BindableEvent") or child:IsA("BindableFunction") then
					log(("    └ [%s] %s"):format(child.ClassName, child:GetFullName()))
				end
			end
		end
	end
end

-- ============================================================
-- HOOK NAMECALL  —  catches RemoteEvent:FireServer / RemoteFunction:InvokeServer
-- ============================================================
local hookSetup = false
local function setupHook()
	if hookSetup then return end
	local ok, mt = pcall(getrawmetatable, game)
	if not ok or not mt then
		log("!! Could not get raw metatable. Hook unavailable.")
		flush(); return
	end
	local setReadonly = setreadonly or make_writeable
	local isReadonly  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end

	if isReadonly(mt) then pcall(setReadonly, mt, false) end

	local oldNamecall = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		local method = getnamecallmethod and getnamecallmethod() or ""
		if method == "FireServer" or method == "InvokeServer"
			or method == "fireServer" or method == "invokeServer" then
			pcall(function()
				log(("[FIRE] %s :%s(%s)"):format(self:GetFullName(), method, serArgs(...)))
			end)
		end
		return oldNamecall(self, ...)
	end)
	pcall(setReadonly, mt, true)
	hookSetup = true
	log("[Recon] Namecall hook installed.")
	flush()
end

setupHook()

-- Wrap log() to bump fire counter when a [FIRE] line is added
local _origLog = log
function log(line)
	if typeof(line) == "string" and line:sub(1, 6) == "[FIRE]" then
		fireCount = fireCount + 1
	end
	_origLog(line)
end
