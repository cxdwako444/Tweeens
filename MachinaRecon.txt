local function g(n)
	local ok, v = pcall(function() return getfenv()[n] end)
	if ok and v then return v end
	return rawget(_G, n)
end

local _buf, _saving = {}, false
local function flush()
	if _saving or #_buf == 0 or not _writefile then return end
	_saving = true
	local chunk = table.concat(_buf, "\n") .. "\n"
	_buf = {}
	pcall(function()
		if _appendfile and _isfile and _isfile(OUT_FILE) then
			_appendfile(OUT_FILE, chunk)
		else
			local prev = ""
			if _readfile and _isfile and _isfile(OUT_FILE) then
				prev = _readfile(OUT_FILE) or ""
			end
			_writefile(OUT_FILE, prev .. chunk)
		end
	end)
	_saving = false
end

local function log(line)
	table.insert(_buf, tostring(line))
	if #_buf >= 20 then flush() end
end

if _writefile then pcall(_writefile, OUT_FILE, "") end
task.spawn(function() while true do task.wait(1.5); flush() end end)

-- NAMECALL HOOK (install before dump so we catch boot-time fires)
-- ============================================================
local function ser(v,d)
	d = d or 0; if d > 3 then return "<deep>" end
	local t = typeof(v)
	if t == "Instance" then
		local cls,fn = "?","?"
		pcall(function() cls = v.ClassName end)
		pcall(function() fn = v:GetFullName() end)
		return ("<%s:%s>"):format(cls,fn)
	elseif t=="Vector3" then return ("V3(%.1f,%.1f,%.1f)"):format(v.X,v.Y,v.Z)
	elseif t=="Vector2" then return ("V2(%.1f,%.1f)"):format(v.X,v.Y)
	elseif t=="CFrame" then local p=v.Position return ("CF(%.1f,%.1f,%.1f)"):format(p.X,p.Y,p.Z)
	elseif t=="Color3" then return ("C3(%d,%d,%d)"):format(v.R*255,v.G*255,v.B*255)
	elseif t=="EnumItem" then return tostring(v)
	elseif t=="string" then return ("%q"):format(v)
	elseif t=="number" or t=="boolean" or t=="nil" then return tostring(v)
	elseif t=="table" then
		local p,n = {},0
		for k,vv in pairs(v) do
			n=n+1; if n>16 then table.insert(p,"...") break end
			table.insert(p, ("[%s]=%s"):format(ser(k,d+1), ser(vv,d+1)))
		end
		return "{"..table.concat(p,", ").."}"
	end
	return ("<%s>"):format(t)
end
local function serArgs(...)
	local n,p = select("#",...), {}
	for i=1,n do p[i] = ser((select(i,...))) end
	return table.concat(p," | ")
end

pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		local method = getnamecallmethod and getnamecallmethod() or ""
		if method == "FireServer" or method == "InvokeServer" then
			fireCount = fireCount + 1
			pcall(function()
				log(("[FIRE #%d] %s :%s(%s)"):format(
					fireCount, self:GetFullName(), method, serArgs(...)))
			end)
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)

-- ============================================================
-- DUMP — runs in BACKGROUND task so UI stays responsive
-- ============================================================
local function flagged(s)
	if type(s) ~= "string" then return false end
	local low = s:lower()
	for _, k in ipairs(KEYWORDS) do
		if low:find(k, 1, true) then return true end
	end
	return false
end

local function tryDecompile(s)
	if not _decompile then return nil, "no-decompile" end
	local done, src, err = false, nil, nil
	task.spawn(function()
		local ok, res = pcall(_decompile, s)
		if ok then src = res else err = tostring(res) end
		done = true
	end)
	local t0 = tick()
	while not done and (tick() - t0) < DECOMPILE_TIME do task.wait(0.05) end
	if not done then return nil, "timeout" end
	if not src or src == "" then return nil, err or "empty" end
	if #src > MAX_SRC_BYTES then
		src = src:sub(1, MAX_SRC_BYTES) .. "\n-- [TRUNCATED] --"
	end
	return src
end

local function logSrc(s, indent)
	indent = indent or ""
	local src, err = tryDecompile(s)
	if not src then
		log(indent .. "  -- <decompile failed: " .. tostring(err) .. ">")
		return
	end
	log(indent .. "  >>>>>>> SOURCE >>>>>>>")
	for line in src:gmatch("([^\n]*)\n?") do log(indent .. line) end
	log(indent .. "  <<<<<<< END SOURCE <<<<<<<")
end

task.spawn(function()
	pcall(function()
		log(("===== MACHINA RECON v2  %s ====="):format(os.date("%Y-%m-%d %H:%M:%S")))
		log(("PlaceId: %d   JobId: %s"):format(game.PlaceId, game.JobId or ""))
		log(("Player:  %s (uid %d)"):format(LocalPlayer.Name, LocalPlayer.UserId))
		log("")
		flush()

		-- 1. REMOTES
		setProgress("[1/4] remote inventory...")
		log(""); log("################ 1. REMOTES ################")
		local function dumpRemotes(parent, label)
			log(""); log("----- " .. label .. " -----")
			local n, fl = 0, 0
			for _, d in ipairs(parent:GetDescendants()) do
				if d:IsA("RemoteEvent") or d:IsA("RemoteFunction")
					or d:IsA("BindableEvent") or d:IsA("BindableFunction")
					or d:IsA("UnreliableRemoteEvent") then
					n = n + 1
					local mk = flagged(d.Name) and "  *** FLAGGED ***" or ""
					if mk ~= "" then fl = fl + 1 end
					log(("[%s] %s%s"):format(d.ClassName, d:GetFullName(), mk))
				end
			end
			log(("(total %d · flagged %d)"):format(n, fl))
		end
		pcall(dumpRemotes, ReplicatedStorage, "ReplicatedStorage")
		pcall(function() dumpRemotes(game:GetService("ReplicatedFirst"), "ReplicatedFirst") end)
		pcall(function() dumpRemotes(workspace, "Workspace") end)
		flush()

		-- 2. TOOLS
		setProgress("[2/4] tools (decompiling)...")
		log(""); log("################ 2. TOOLS ################")
		local function dumpTool(t, where)
			log(""); log(("----- Tool [%s] %s -----"):format(where, t.Name))
			log("Path: " .. t:GetFullName())
			for k, v in pairs(t:GetAttributes()) do
				log(("  @%s = %s"):format(k, ser(v)))
			end
			for _, child in ipairs(t:GetDescendants()) do
				local cls = child.ClassName
				if child:IsA("LocalScript") or child:IsA("ModuleScript") or child:IsA("Script") then
					log(("-- script: [%s] %s"):format(cls, child:GetFullName()))
					pcall(logSrc, child, "  ")
				elseif child:IsA("RemoteEvent") or child:IsA("RemoteFunction")
					or child:IsA("BindableEvent") or child:IsA("BindableFunction") then
					log(("-- remote: [%s] %s"):format(cls, child:GetFullName()))
				end
			end
		end
		local function dumpToolsIn(parent, label)
			if not parent then return end
			for _, c in ipairs(parent:GetChildren()) do
				if c:IsA("Tool") then pcall(dumpTool, c, label) end
			end
		end
		dumpToolsIn(LocalPlayer:FindFirstChild("Backpack"), "Backpack")
		dumpToolsIn(StarterPack, "StarterPack")
		if LocalPlayer.Character then dumpToolsIn(LocalPlayer.Character, "Character") end
		flush()

		-- 3. PLAYERGUI
		setProgress("[3/4] PlayerGui scripts...")
		log(""); log("################ 3. PLAYERGUI ################")
		for _, gnode in ipairs(PlayerGui:GetChildren()) do
			if gnode.Name == "MachinaReconFull" then continue end
			if gnode:IsA("ScreenGui") or gnode:IsA("Frame") then
				pcall(function()
					log(""); log(("----- ScreenGui: %s -----"):format(gnode.Name))
					log("Path: " .. gnode:GetFullName())
					for _, d in ipairs(gnode:GetDescendants()) do
						if d:IsA("LocalScript") or d:IsA("ModuleScript") or d:IsA("Script") then
							log(("-- script: [%s] %s"):format(d.ClassName, d:GetFullName()))
							pcall(logSrc, d, "  ")
						elseif d:IsA("TextButton") or d:IsA("ImageButton") then
							local txt = d:IsA("TextButton") and tostring(d.Text) or ""
							if flagged(d.Name) or flagged(txt) then
								log(("-- button *** [%s] text=%q  %s"):format(d.ClassName, txt, d:GetFullName()))
							end
						end
					end
				end)
			end
		end
		flush()

		-- 4. MODULES
		setProgress("[4/4] mining modules...")
		log(""); log("################ 4. MINING MODULES ################")
		local seen = {}
		local dumpedCount = 0
		local function maybe(m, why)
			if dumpedCount >= MAX_MODULES then return end
			if seen[m] then return end; seen[m] = true
			if not (m:IsA("ModuleScript") or m:IsA("LocalScript") or m:IsA("Script")) then return end
			local path = ""
			pcall(function() path = m:GetFullName() end)
			if not (flagged(m.Name) or flagged(path)) then return end
			dumpedCount = dumpedCount + 1
			setProgress(("[4/4] mining modules... %d/%d"):format(dumpedCount, MAX_MODULES))
			pcall(function()
				log(""); log(("----- [%s] %s (%s) -----"):format(m.ClassName, path, why))
				logSrc(m, "")
			end)
		end
		pcall(function()
			for _, d in ipairs(ReplicatedStorage:GetDescendants()) do maybe(d, "RS") end
		end)
		pcall(function()
			for _, d in ipairs(StarterPlayer:GetDescendants()) do maybe(d, "StarterPlayer") end
		end)
		pcall(function()
			local ps = LocalPlayer:FindFirstChild("PlayerScripts")
			if ps then for _, d in ipairs(ps:GetDescendants()) do maybe(d, "PlayerScripts") end end
		end)
		if _getloaded then
			pcall(function()
				for _, m in ipairs(_getloaded()) do maybe(m, "loaded") end
			end)
		end
		if _getscripts then
			pcall(function()
				for _, m in ipairs(_getscripts()) do maybe(m, "scripts") end
			end)
		end
		flush()

		log(""); log("################ READY — USE THE MARKER BUTTONS ################")
		flush()
		setProgress(("READY · %d fires logged"):format(fireCount))
	end)
end)
