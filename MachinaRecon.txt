-- File:   MachinaRecon.txt
--
-- WHAT v9 GOT WRONG:
--   We iterated workspace.CollisionParts. Problem: mined blocks are
--   removed from that folder immediately. After mining the first
--   ~10 blocks, getChildren() returned empty and we stopped firing.
--   That's why you mined 10 blocks then nothing.
--
-- v10 FIX:
--   Generate target positions from a GRID centered on you, snapped
--   to (12000,12000,12000) origin + 6-stud spacing. Doesn't depend
--   on parts existing. Keeps hitting every grid cell in your radius
--   so as soon as a block appears/respawns/streams in, it gets hit.
--
-- DEFAULTS (safe to bump):
--   Radius=12, Hz=10, Hits/blk=1, Damage=1
--   Total fires/sec ~= (cells_in_sphere) * Hz * hits
--   For r=12: 65 cells * 10Hz * 1 = 650 fires/sec
--
-- WATCH "MINED/sec" in the status to tune empirically.
-- ============================================================

local ORIGIN     = Vector3.new(12000, 12000, 12000)
local BLOCK_SIZE = 6

-- ============================================================
-- ENV
-- ============================================================
local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end
local _writefile = find("writefile")

local LOG = {}
local function log(s)
	table.insert(LOG, tostring(s))
	if #LOG > 200 then table.remove(LOG, 1) end
end

-- ============================================================
-- BUFFER + FIRE
-- ============================================================
local function makeMineBuffer(x, y, z, damage)
	local b = buffer.create(32)
	buffer.writef64(b, 0,  (z - ORIGIN.Z) / BLOCK_SIZE)
	buffer.writef64(b, 8,  (y - ORIGIN.Y) / BLOCK_SIZE)
	buffer.writef64(b, 16, (x - ORIGIN.X) / BLOCK_SIZE)
	buffer.writef64(b, 24, damage)
	return b
end
local function randomNonce()
	return string.char(
		math.random(0,255), math.random(0,255),
		math.random(0,255), math.random(0,255))
end
local function fireMine(x, y, z, dmg)
	pcall(function()
		R:FireServer(randomNonce(), "DamageBlockSmooth",
			makeMineBuffer(x, y, z, dmg))
	end)
end

-- ============================================================
-- GRID TARGET GENERATION  (the real fix)
-- Snap player position to (12000 + k*6) grid cells, then enumerate
-- every cell within `radius` studs.
-- ============================================================
local function snap(coord, origin, size)
	return math.floor((coord - origin) / size + 0.5) * size + origin
end

local function generateGridTargets(playerPos, radius)
	local cx = snap(playerPos.X, ORIGIN.X, BLOCK_SIZE)
	local cy = snap(playerPos.Y, ORIGIN.Y, BLOCK_SIZE)
	local cz = snap(playerPos.Z, ORIGIN.Z, BLOCK_SIZE)
	local n = math.ceil(radius / BLOCK_SIZE)
	local r2 = radius * radius
	local targets = {}
	for dx = -n, n do
		local ox = dx * BLOCK_SIZE
		for dy = -n, n do
			local oy = dy * BLOCK_SIZE
			for dz = -n, n do
				local oz = dz * BLOCK_SIZE
				local d2 = ox*ox + oy*oy + oz*oz
				if d2 <= r2 then
					table.insert(targets, {
						x = cx + ox, y = cy + oy, z = cz + oz, d2 = d2,
					})
				end
			end
		end
	end
	table.sort(targets, function(a,b) return a.d2 < b.d2 end)
	return targets
end

-- ============================================================
-- STATE
-- ============================================================
local stats = {
	radiusOn = false,
	radius = 12,
	rateHz = 10,
	hitsPerBlock = 1,
	damage = 1.0,
	totalFires = 0,
	firesThisSec = 0,
	firesPerSec = 0,
	minedThisSec = 0,
	minedPerSec = 0,
	totalMined = 0,
	gridCells = 0,
}

-- ============================================================
-- TRACK BLOCKS REMOVED (= server confirmed mined)
-- We watch workspace.Blocks (which doesn't randomly stream out)
-- ============================================================
local function attachWatcher(folder)
	if not folder then return end
	folder.DescendantRemoving:Connect(function(c)
		if c:IsA("BasePart") then
			stats.minedThisSec = stats.minedThisSec + 1
			stats.totalMined = stats.totalMined + 1
		end
	end)
end
attachWatcher(Workspace:FindFirstChild("CollisionParts"))
attachWatcher(Workspace:FindFirstChild("Blocks"))
Workspace.ChildAdded:Connect(function(c)
	if c.Name == "CollisionParts" or c.Name == "Blocks" then attachWatcher(c) end
end)

task.spawn(function()
	while true do
		task.wait(1.0)
		stats.minedPerSec = stats.minedThisSec
		stats.minedThisSec = 0
		stats.firesPerSec = stats.firesThisSec
		stats.firesThisSec = 0
	end
end)

-- ============================================================
-- RADIUS MINE LOOP
-- ============================================================
task.spawn(function()
	while true do
		if stats.radiusOn then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local targets = generateGridTargets(hrp.Position, stats.radius)
				stats.gridCells = #targets
				for hit = 1, stats.hitsPerBlock do
					for _, t in ipairs(targets) do
						fireMine(t.x, t.y, t.z, stats.damage)
						stats.totalFires = stats.totalFires + 1
						stats.firesThisSec = stats.firesThisSec + 1
					end
				end
			else
				stats.gridCells = 0
			end
			task.wait(1 / math.max(1, stats.rateHz))
		else
			task.wait(0.1)
		end
	end
end)

local function pillRow(y, label, values, getter, setter)
	local lbl = Instance.new("TextLabel")
	lbl.Size=UDim2.new(0,55,0,20); lbl.Position=UDim2.new(0,6,0,y)

	lbl.TextXAlignment=Enum.TextXAlignment.Left; lbl.Parent=root
	local boxes = {}
	local total = 270 - 70
	for i, v in ipairs(values) do
		local pillW = (total / #values) - 4
		local b = Instance.new("TextButton")
		b.Size=UDim2.new(0,pillW,0,20)
		b.Position=UDim2.new(0, 65+(i-1)*(pillW+4), 0, y)
		b.BackgroundColor3 = (getter() == v) and C.BLUE or C.CARD
		b.AutoButtonColor=false; b.Text=tostring(v); b.Font=Enum.Font.GothamBold
		b.TextSize=9; b.TextColor3=C.TXT; b.Parent=root; corner(b,4)

			setter(v)
			for j, bb in ipairs(boxes) do
				bb.BackgroundColor3 = (values[j] == v) and C.BLUE or C.CARD
			end
		end)
		boxes[i] = b
	end
end

	titleBar.InputBegan:Connect(function(input)

	end)
