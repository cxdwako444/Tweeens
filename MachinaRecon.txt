--   - Defaults bumped: hz=20, hits=3, dmg=1

--   - Bump Damage to 5 or 100 to test if server clamps damage

-- ============================================================
-- ENV
-- ============================================================
local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end
local _writefile = find("writefile")

-- ============================================================
-- LOG
-- ============================================================
local LOG = {}
local LOG_MAX = 200
local function log(s)
	table.insert(LOG, tostring(s))
	if #LOG > LOG_MAX then table.remove(LOG, 1) end
end

-- ============================================================
-- BUFFER + FIRE
-- ============================================================
local function makeMineBuffer(worldPos, damage)
	local b = buffer.create(32)
	buffer.writef64(b, 0,  (worldPos.Z - ORIGIN.Z) / BLOCK_SIZE)
	buffer.writef64(b, 8,  (worldPos.Y - ORIGIN.Y) / BLOCK_SIZE)
	buffer.writef64(b, 16, (worldPos.X - ORIGIN.X) / BLOCK_SIZE)
	buffer.writef64(b, 24, damage)
	return b
end
local function randomNonce()
	return string.char(
		math.random(0,255), math.random(0,255),
		math.random(0,255), math.random(0,255))
end
local function fireMine(worldPos, dmg)
	pcall(function()
		R:FireServer(randomNonce(), "DamageBlockSmooth",
			makeMineBuffer(worldPos, dmg))
	end)
end

local stats = {
	totalFires = 0,
	radiusOn = false,
	radius = 12,
	rateHz = 20,
	damage = 1.0,
	hitsPerBlock = 3,
	blocksLastCycle = 0,
	minedPerSec = 0,
	minedThisSec = 0,
}

-- ============================================================
-- TRACK MINED BLOCKS (CollisionParts.ChildRemoved)
-- ============================================================
local removalConn
local function attachMineCounter()
	if removalConn then removalConn:Disconnect() end
	local cp = Workspace:FindFirstChild("CollisionParts")
	if not cp then return end
	removalConn = cp.ChildRemoved:Connect(function(c)
		if c:IsA("BasePart") then
			stats.minedThisSec = stats.minedThisSec + 1
		end
	end)
end
attachMineCounter()
Workspace.ChildAdded:Connect(function(c)
	if c.Name == "CollisionParts" then attachMineCounter() end
end)

task.spawn(function()
	while true do
		task.wait(1.0)
		stats.minedPerSec = stats.minedThisSec
		stats.minedThisSec = 0
	end
end)

-- ============================================================
-- RADIUS MINE LOOP
-- ============================================================
local function getSortedNearby(origin, radius)
	local cp = Workspace:FindFirstChild("CollisionParts")
	if not cp then return {} end
	local r2 = radius * radius
	local out = {}
	for _, d in ipairs(cp:GetChildren()) do
		if d:IsA("BasePart") then
			local dp = d.Position - origin
			local d2 = dp.X*dp.X + dp.Y*dp.Y + dp.Z*dp.Z
			if d2 <= r2 then
				table.insert(out, {pos=d.Position, d2=d2})
			end
		end
	end
	table.sort(out, function(a,b) return a.d2 < b.d2 end)
	return out
end

task.spawn(function()
	while true do
		if stats.radiusOn then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local targets = getSortedNearby(hrp.Position, stats.radius)
				stats.blocksLastCycle = #targets
				-- Fire each block N times per cycle (closest first)
				for hit = 1, stats.hitsPerBlock do
					for _, t in ipairs(targets) do
						fireMine(t.pos, stats.damage)
						stats.totalFires = stats.totalFires + 1
					end
				end
			else
				stats.blocksLastCycle = 0
			end
			task.wait(1 / math.max(1, stats.rateHz))
		else
			task.wait(0.1)
		end
	end
end)
