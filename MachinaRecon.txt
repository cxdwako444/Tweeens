local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end

-- ============================================================
-- OUTPUT (no game print — anti-cheat reads it)

local OUTPUT = {}
local function tstamp() return os.date("%H:%M:%S") end
local function dprint(...)
	local parts = {}
	for i = 1, select("#", ...) do parts[i] = tostring((select(i, ...))) end
	table.insert(OUTPUT, ("[%s] %s"):format(tstamp(), table.concat(parts, " ")))
	if #OUTPUT > 400 then table.remove(OUTPUT, 1) end
end

-- ============================================================
-- HELPERS
-- ============================================================
local function dump(v, depth)
	depth = depth or 0
	if depth > 3 then return "<deep>" end
	local t = typeof(v)
	if t == "Instance" then
		local fn = "?"; pcall(function() fn = v:GetFullName() end)
		return ("<%s:%s>"):format(v.ClassName, fn)
	elseif t == "Vector3" then return ("V3(%.1f,%.1f,%.1f)"):format(v.X, v.Y, v.Z)
	elseif t == "buffer" then
		local ok, raw = pcall(buffer.tostring, v)
		if ok then
			local hex = {}; local lim = math.min(#raw, 48)
			for i = 1, lim do hex[i] = string.format("%02x", string.byte(raw, i)) end
			return ("buf(%d) %s"):format(#raw, table.concat(hex, " "))
		end
		return "buf<?>"
	elseif t == "string" then return ('"%s"'):format(v:sub(1, 50))
	elseif t == "table" then
		local parts, n = {}, 0
		for k, vv in pairs(v) do
			n = n + 1; if n > 12 then table.insert(parts, "..."); break end
			table.insert(parts, ("[%s]=%s"):format(tostring(k), dump(vv, depth+1)))
		end
		return "{" .. table.concat(parts, ", ") .. "}"
	end
	return tostring(v)
end

local function rnonce()
	return string.char(math.random(0,255), math.random(0,255), math.random(0,255), math.random(0,255))
end

local function snapCoord(c, origin) return math.floor((c - origin) / BLOCK_SIZE + 0.5) * BLOCK_SIZE + origin end
local function mineBuf(wx, wy, wz, dmg)
	local b = buffer.create(32)
	buffer.writef64(b, 0,  (snapCoord(wz, ORIGIN.Z) - ORIGIN.Z) / BLOCK_SIZE)
	buffer.writef64(b, 8,  (snapCoord(wy, ORIGIN.Y) - ORIGIN.Y) / BLOCK_SIZE)
	buffer.writef64(b, 16, (snapCoord(wx, ORIGIN.X) - ORIGIN.X) / BLOCK_SIZE)
	buffer.writef64(b, 24, dmg or 0.1)
	return b
end
local function fireMine(x, y, z, dmg)
	pcall(function() R:FireServer(rnonce(), "DamageBlockSmooth", mineBuf(x, y, z, dmg)) end)
end
local function nearest()
	local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not hrp then return nil end
	local cp = Workspace:FindFirstChild("CollisionParts")
	if not cp then return nil end
	local best, bd = nil, math.huge
	for _, p in ipairs(cp:GetChildren()) do
		if p:IsA("BasePart") then
			local d = (p.Position - hrp.Position).Magnitude
			if d < bd then bd = d; best = p end
		end
	end
	return best and best.Position or nil
end

-- Count nearby blocks that disappear in a time window (for explode test)
local function countMinesDuring(seconds)
	local cp = Workspace:FindFirstChild("CollisionParts")
	if not cp then return 0 end
	local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not hrp then return 0 end
	local count = 0
	local conn = cp.ChildRemoved:Connect(function(c)
		if c:IsA("BasePart") then
			local d = (c.Position - hrp.Position).Magnitude
			if d <= 40 then count = count + 1 end
		end
	end)
	task.wait(seconds)
	conn:Disconnect()
	return count
end

-- ============================================================
-- SNIFFER
-- ============================================================
local sniffOn = false
pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		if sniffOn then
			local method = getnamecallmethod and getnamecallmethod() or ""
			if method == "FireServer" or method == "InvokeServer" then
				local n, parts = select("#", ...), {}
				for i = 1, n do parts[i] = dump((select(i, ...))) end
				pcall(function()
					table.insert(OUTPUT, ("[%s] SNIFF %s:%s(%s)"):format(
						tstamp(), self:GetFullName(), method, table.concat(parts, ", ")))
				end)
			end
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)

local function makeEnv()
	return setmetatable({
		dprint = dprint, dump = dump, rnonce = rnonce,
		mineBuf = mineBuf, fireMine = fireMine, nearest = nearest,
		countMinesDuring = countMinesDuring,
		R = R, DE = DE, RS = ReplicatedStorage, WS = Workspace,
		LP = LocalPlayer,
		CHAR = LocalPlayer.Character,
		HRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart"),
		NONCE = MAGIC_NONCE, ORIGIN = ORIGIN, BLOCK_SIZE = BLOCK_SIZE,
	}, {__index = getfenv()})
end
local function runCode(code)
	local fn, err = loadstring(code, "console")
	if not fn then dprint("PARSE ERR: " .. tostring(err)); return end
	setfenv(fn, makeEnv())
	dprint("--- running ---")
	local ok, result = pcall(fn)
	if not ok then dprint("RUN ERR: " .. tostring(result))
	elseif result ~= nil then dprint("=> " .. dump(result)) end
	dprint("--- done ---")
end
