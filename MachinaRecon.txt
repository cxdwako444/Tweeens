-- ============================================================
-- MACHINA RECON v11  —  REAL TARGETS + ALL-VISIBLE MODE
-- Branch: claude/rare-indicator-auto-aim-mC8LF
-- File:   MachinaRecon.txt
--
-- WHY v10 WAS SLOWER:
--   v10 grid-spammed every cell in radius. The server validates
--   that a block exists at each (Z,Y,X) before mining. Empty cells
--   = wasted fires that count toward rate-limit. We were drowning
--   legit hits in noise.
--
-- v11 STRATEGY:
--   - Only fire at REAL parts (union of workspace.CollisionParts
--     + workspace.Blocks descendants) — these have actual block
--     positions the server will accept.
--   - Two modes:
--       Near  — only parts within radius (default)
--       All   — every visible mineable part the client knows about
--               (use when local area runs dry)
--   - Snap positions to the (12000, 6) grid before firing, so the
--     buffer values land on exact integer block coords.
--   - Bigger stats display so you can tune empirically.
-- ============================================================

local Players          = game:GetService("Players")

local OUT_FILE    = "MachinaRecon.txt"
local MAGIC_NONCE = string.char(255,255,255,255)
local R = ReplicatedStorage:WaitForChild("RemoteEvent")

local ORIGIN     = Vector3.new(12000, 12000, 12000)
local BLOCK_SIZE = 6

-- ============================================================
-- ENV
-- ============================================================
local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end
local _writefile = find("writefile")

-- ============================================================
-- BUFFER + FIRE  (with grid snapping)
-- ============================================================
local function snapCoord(c, origin)
	return math.floor((c - origin) / BLOCK_SIZE + 0.5) * BLOCK_SIZE + origin
end

local function makeMineBuffer(worldX, worldY, worldZ, damage)
	local b = buffer.create(32)
	-- Snap to integer block coords before division (avoids float drift)
	local bz = (snapCoord(worldZ, ORIGIN.Z) - ORIGIN.Z) / BLOCK_SIZE
	local by = (snapCoord(worldY, ORIGIN.Y) - ORIGIN.Y) / BLOCK_SIZE
	local bx = (snapCoord(worldX, ORIGIN.X) - ORIGIN.X) / BLOCK_SIZE
	buffer.writef64(b, 0,  bz)
	buffer.writef64(b, 8,  by)
	buffer.writef64(b, 16, bx)
	buffer.writef64(b, 24, damage)
	return b
end

local function randomNonce()
	return string.char(
		math.random(0,255), math.random(0,255),
		math.random(0,255), math.random(0,255))
end

local function fireMine(x, y, z, dmg)
	pcall(function()
		R:FireServer(randomNonce(), "DamageBlockSmooth",
			makeMineBuffer(x, y, z, dmg))
	end)
end

-- ============================================================
-- COLLECT TARGETS  (CollisionParts ∪ Blocks-descendants)
-- ============================================================
local function collectTargets(playerPos, radius, mode)
	local r2 = radius * radius
	local seen = {}  -- dedupe by grid-snapped key
	local result = {}

	local function tryAdd(part)
		if not part:IsA("BasePart") then return end
		local p = part.Position
		if mode == "Near" then
			local dp = p - playerPos
			if dp.X*dp.X + dp.Y*dp.Y + dp.Z*dp.Z > r2 then return end
		end
		-- Snap to grid for dedupe
		local kx = math.floor((p.X - ORIGIN.X) / BLOCK_SIZE + 0.5)
		local ky = math.floor((p.Y - ORIGIN.Y) / BLOCK_SIZE + 0.5)
		local kz = math.floor((p.Z - ORIGIN.Z) / BLOCK_SIZE + 0.5)
		local key = kx .. "," .. ky .. "," .. kz
		if seen[key] then return end
		seen[key] = true
		table.insert(result, {x=p.X, y=p.Y, z=p.Z})
	end

	-- CollisionParts (server-recognized mineables)
	local cp = Workspace:FindFirstChild("CollisionParts")
	if cp then
		for _, d in ipairs(cp:GetChildren()) do tryAdd(d) end
	end

	-- Blocks folder (visual layer; persists longer than CP)
	local bl = Workspace:FindFirstChild("Blocks")
	if bl then
		for _, child in ipairs(bl:GetChildren()) do
			if child:IsA("BasePart") then
				tryAdd(child)
			elseif child:IsA("Folder") or child:IsA("Model") then
				for _, gc in ipairs(child:GetChildren()) do tryAdd(gc) end
			end
		end
	end

	return result
end

-- ============================================================
-- STATE
-- ============================================================
local stats = {
	radiusOn = false,
	mode = "Near",          -- "Near" or "All"
	radius = 12,
	rateHz = 10,
	damage = 1.0,
	totalFires = 0,
	firesThisSec = 0,
	firesPerSec = 0,
	minedThisSec = 0,
	minedPerSec = 0,
	totalMined = 0,
	targetsFound = 0,
}

-- Track mined blocks (anything removed from Blocks or CollisionParts)
local function attachWatcher(folder)
	if not folder then return end
	folder.DescendantRemoving:Connect(function(c)
		if c:IsA("BasePart") then
			stats.minedThisSec = stats.minedThisSec + 1
			stats.totalMined = stats.totalMined + 1
		end
	end)
end
attachWatcher(Workspace:FindFirstChild("CollisionParts"))
attachWatcher(Workspace:FindFirstChild("Blocks"))
Workspace.ChildAdded:Connect(function(c)
	if c.Name == "CollisionParts" or c.Name == "Blocks" then attachWatcher(c) end
end)

task.spawn(function()
	while true do
		task.wait(1.0)
		stats.minedPerSec = stats.minedThisSec; stats.minedThisSec = 0
		stats.firesPerSec = stats.firesThisSec; stats.firesThisSec = 0
	end
end)

-- ============================================================
-- LOOP
-- ============================================================
task.spawn(function()
	while true do
		if stats.radiusOn then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local targets = collectTargets(hrp.Position, stats.radius, stats.mode)
				stats.targetsFound = #targets
				for _, t in ipairs(targets) do
					fireMine(t.x, t.y, t.z, stats.damage)
					stats.totalFires = stats.totalFires + 1
					stats.firesThisSec = stats.firesThisSec + 1
				end
			else
				stats.targetsFound = 0
			end
			task.wait(1 / math.max(1, stats.rateHz))
		else
			task.wait(0.1)
		end
	end
end)

local function pillRow(y, label, values, getter, setter, fmt)
	fmt = fmt or tostring
	local lbl = Instance.new("TextLabel")
	lbl.Size=UDim2.new(0,55,0,20); lbl.Position=UDim2.new(0,6,0,y)
	lbl.BackgroundTransparency=1; lbl.Text=label
	lbl.Font=Enum.Font.GothamBold; lbl.TextSize=9; lbl.TextColor3=C.DIM
	lbl.TextXAlignment=Enum.TextXAlignment.Left; lbl.Parent=root
	local boxes = {}
	local total = 270 - 70
	for i, v in ipairs(values) do
		local pillW = (total / #values) - 4
		local b = Instance.new("TextButton")
		b.Size=UDim2.new(0,pillW,0,20)
		b.Position=UDim2.new(0, 65+(i-1)*(pillW+4), 0, y)
		b.BackgroundColor3 = (getter() == v) and C.BLUE or C.CARD
		b.AutoButtonColor=false; b.Text=fmt(v); b.Font=Enum.Font.GothamBold
		b.TextSize=9; b.TextColor3=C.TXT; b.Parent=root; corner(b,4)
		b.MouseButton1Click:Connect(function()
			setter(v)
			for j, bb in ipairs(boxes) do
				bb.BackgroundColor3 = (values[j] == v) and C.BLUE or C.CARD
			end
		end)
		boxes[i] = b
	end
end

do
	local dragging, ds, sp_ = false, nil, nil
	titleBar.InputBegan:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1
			or input.UserInputType==Enum.UserInputType.Touch then
			dragging=true; ds=input.Position; sp_=root.Position
			input.Changed:Connect(function()
				if input.UserInputState==Enum.UserInputState.End then dragging=false end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType~=Enum.UserInputType.MouseMovement
			and input.UserInputType~=Enum.UserInputType.Touch then return end
		local d = input.Position - ds
		root.Position = UDim2.new(sp_.X.Scale, sp_.X.Offset+d.X,
			sp_.Y.Scale, sp_.Y.Offset+d.Y)
	end)
end
