-- ============================================================

local ORIGIN     = Vector3.new(12000, 12000, 12000)
local BLOCK_SIZE = 6

-- ============================================================
-- ENV
-- ============================================================
local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end
local _writefile = find("writefile")

-- ============================================================
-- LOG
-- ============================================================
local LOG = {}
local LOG_MAX = 250
local function log(s)
	table.insert(LOG, tostring(s))
	if #LOG > LOG_MAX then table.remove(LOG, 1) end
end

-- ============================================================
-- BUFFER BUILDER  (the decoded formula)
-- ============================================================
local function makeMineBuffer(worldPos, damage)
	local b = buffer.create(32)
	buffer.writef64(b, 0,  (worldPos.Z - ORIGIN.Z) / BLOCK_SIZE)
	buffer.writef64(b, 8,  (worldPos.Y - ORIGIN.Y) / BLOCK_SIZE)
	buffer.writef64(b, 16, (worldPos.X - ORIGIN.X) / BLOCK_SIZE)
	buffer.writef64(b, 24, damage)
	return b
end

local function randomNonce()
	return string.char(
		math.random(0,255), math.random(0,255),
		math.random(0,255), math.random(0,255))
end

local function fireMine(worldPos, dmg)
	pcall(function()
		R:FireServer(randomNonce(), "DamageBlockSmooth",
			makeMineBuffer(worldPos, dmg))
	end)
end

-- ============================================================
-- HEX
-- ============================================================
local function strHex(s, max)
	max = max or 64; local n = math.min(#s, max); local out = {}
	for i = 1, n do out[i] = string.format("%02x", string.byte(s, i)) end
	local h = table.concat(out, " ")
	if #s > n then h = h .. (" +%d"):format(#s - n) end
	return h
end
local function bufHex(b, max)
	local ok, raw = pcall(buffer.tostring, b)
	if ok and type(raw) == "string" then return strHex(raw, max or 200), #raw end
	return "<no>", 0
end

local stats = {

-- ============================================================
-- HOOK
-- ============================================================
pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		if stats.hookEnabled then
			local method = getnamecallmethod and getnamecallmethod() or ""
			if method == "FireServer" or method == "InvokeServer" then
				stats.totalFires = stats.totalFires + 1
				local a2 = select(2, ...)
				if type(a2) == "string" and a2 == "DamageBlockSmooth" then
					stats.mineFires = stats.mineFires + 1
					if stats.captureRemaining > 0 then
						stats.captureRemaining = stats.captureRemaining - 1
						local a1 = select(1, ...)
						local a3 = select(3, ...)
						local nhex = type(a1)=="string" and strHex(a1, 8) or "?"
						local hex, len = "?", 0
						if typeof(a3) == "buffer" then hex, len = bufHex(a3, 256) end
						log(("[CAP%d] %s len=%d %s"):format(
							stats.mineFires, nhex, len, hex))
					end
				end
			end
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)

-- ============================================================
-- RADIUS MINE LOOP
-- ============================================================
local function getNearbyMineable(origin, radius)
	local cp = Workspace:FindFirstChild("CollisionParts")
	if not cp then return {} end
	local r2 = radius * radius
	local out = {}
	for _, d in ipairs(cp:GetChildren()) do
		if d:IsA("BasePart") then
			local dp = d.Position - origin
			if dp.X*dp.X + dp.Y*dp.Y + dp.Z*dp.Z <= r2 then
				table.insert(out, d.Position)
			end
		end
	end
	return out
end

task.spawn(function()
	while true do
		if stats.radiusOn then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local positions = getNearbyMineable(hrp.Position, stats.radius)
				stats.radiusBlocksLastCycle = #positions
				for _, p in ipairs(positions) do
					fireMine(p, stats.damage)
				end
			else
				stats.radiusBlocksLastCycle = 0
			end
			task.wait(1 / math.max(1, stats.rateHz))
		else
			task.wait(0.1)
		end
	end
end)
