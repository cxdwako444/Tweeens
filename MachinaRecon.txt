local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end

local LOG_LINES = {}
local LOG_MAX = 400
local function log(s)
	s = tostring(s)
	table.insert(LOG_LINES, s)
	if #LOG_LINES > LOG_MAX then table.remove(LOG_LINES, 1) end
	print("[R4] " .. s)
end

local function logHeader()

		tostring(_writefile~=nil), tostring(_readfile~=nil), tostring(_appendfile~=nil)))
	log(("Player: %s  PlaceId: %d"):format(LocalPlayer.Name, game.PlaceId))
end

-- ============================================================
-- BYTE HELPERS
-- ============================================================
local function strHex(s, max)
	max = max or 64
	local n = math.min(#s, max)
	local out = {}
	for i = 1, n do out[i] = string.format("%02x", string.byte(s, i)) end
	local hex = table.concat(out, " ")
	if #s > n then hex = hex .. (" ...+%d"):format(#s - n) end
	return hex
end
local function bufHex(b, max)
	max = max or 96
	local ok, raw = pcall(buffer.tostring, b)
	if ok and type(raw) == "string" then return strHex(raw, max), #raw end
	return "<unreadable>", 0
end

-- Serialize ONE arg compactly. No deep tables, no work for instances.
local function sarg(v)
	local t = typeof(v)
	if t == "string" then
		local p = v:gsub("[%c\128-\255]", ".")
		return ("\"%s\"(hex=%s)"):format(p, strHex(v, 8))
	elseif t == "buffer" then
		local hex, n = bufHex(v); return ("<buf len=%d %s>"):format(n, hex)
	elseif t == "Instance" then
		local fn = ""; pcall(function() fn = v:GetFullName() end); return "<"..t..":"..fn..">"
	elseif t == "Vector3" then return ("V3(%.0f,%.0f,%.0f)"):format(v.X,v.Y,v.Z)
	elseif t == "boolean" or t == "number" or t == "nil" then return tostring(v)
	elseif t == "table" then return "<table>"  -- skip; too risky for perf
	else return "<"..t..">" end
end

local stats = {
	totalFires = 0,
	captureRemaining = 0,
	hookEnabled = true,
	markCount = 0,
	cmds = {},  -- [commandName] = count
}

-- ============================================================
-- HOOK — minimal work by default
-- ============================================================
pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		if stats.hookEnabled then
			local method = getnamecallmethod and getnamecallmethod() or ""
			if method == "FireServer" or method == "InvokeServer" then
				stats.totalFires = stats.totalFires + 1
				local args = {...}
				local cmd = type(args[2]) == "string" and args[2] or "?"
				stats.cmds[cmd] = (stats.cmds[cmd] or 0) + 1
				if stats.captureRemaining > 0 then
					stats.captureRemaining = stats.captureRemaining - 1
					-- ONLY here do we do heavy serialization
					local parts = {}
					for i = 1, select("#", ...) do parts[i] = sarg((select(i, ...))) end
					log(("[CAP %d] cmd=%s args=%s"):format(
						stats.totalFires, cmd, table.concat(parts, " | ")))
				end
			end
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)
logHeader()
log("hook installed (idle — taps Capture to log args)")

fileBtn = btn(202, "💾 Test Write + Save Log", C.PURPLE, function()
	if not _writefile then
		log("ERR: writefile() does not exist in this exec env")
		fileBtn.Text = "✗ no writefile"; return
	end
	local payload = table.concat(LOG_LINES, "\n")
	local ok, err = pcall(_writefile, OUT_FILE, payload)
	if not ok then
		log("ERR writefile: " .. tostring(err))
		fileBtn.Text = "✗ write FAILED"; return
	end
	-- Read it back to verify
	local back = nil
	if _readfile then
		local ok2, r = pcall(_readfile, OUT_FILE)
		if ok2 then back = r end
	end
	if back == payload then
		log(("✓ wrote %d bytes to %s, verified"):format(#payload, OUT_FILE))
		fileBtn.Text = ("✓ saved %d bytes"):format(#payload)
	elseif back then
		log(("? wrote but readback differs (read %d bytes)"):format(#back))
		fileBtn.Text = "? wrote but mismatch"
	else
		log(("? wrote but cannot read back"))
		fileBtn.Text = "? wrote, no read"
	end
	task.delay(3, function() if fileBtn.Parent then fileBtn.Text = "💾 Test Write + Save Log" end end)
end)
