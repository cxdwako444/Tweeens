local function g(n)
	local ok, v = pcall(function() return getfenv()[n] end)
	if ok and v then return v end
	return rawget(_G, n)
end
local _writefile  = g("writefile")
local _appendfile = g("appendfile")
local _readfile   = g("readfile")
local _isfile     = g("isfile")

local _buf, _saving = {}, false
local function flush()
	if _saving or #_buf == 0 or not _writefile then return end
	_saving = true
	local chunk = table.concat(_buf, "\n") .. "\n"
	_buf = {}
	pcall(function()
		if _appendfile and _isfile and _isfile(OUT_FILE) then
			_appendfile(OUT_FILE, chunk)
		else
			local prev = ""
			if _readfile and _isfile and _isfile(OUT_FILE) then
				prev = _readfile(OUT_FILE) or ""
			end
			_writefile(OUT_FILE, prev .. chunk)
		end
	end)
	_saving = false
end

local function log(line)
	table.insert(_buf, tostring(line))
	if #_buf >= 20 then flush() end
end

if _writefile then pcall(_writefile, OUT_FILE, "") end
task.spawn(function() while true do task.wait(1.5); flush() end end)

-- HEX HELPERS
-- ============================================================
local function bytesToHex(s, max)
	max = max or #s
	local n = math.min(#s, max)
	local out = {}
	for i = 1, n do
		out[i] = string.format("%02x", string.byte(s, i))
	end
	local hex = table.concat(out, " ")
	if #s > n then hex = hex .. " ... (+" .. (#s - n) .. " more)" end
	return hex
end

local function bufferToHex(b, max)
	-- Roblox 'buffer' type. Try buffer.tostring first.
	local ok, raw = pcall(buffer.tostring, b)
	if ok and type(raw) == "string" then
		return bytesToHex(raw, max), #raw
	end
	-- Fallback: byte-by-byte
	local len = 0
	pcall(function() len = buffer.len(b) end)
	if len == 0 then return "<unreadable buffer>", 0 end
	local n = math.min(len, max)
	local out = {}
	for i = 0, n - 1 do
		local ok2, byte = pcall(buffer.readu8, b, i)
		out[i+1] = string.format("%02x", ok2 and byte or 0)
	end
	local hex = table.concat(out, " ")
	if len > n then hex = hex .. " ... (+" .. (len - n) .. " more)" end
	return hex, len
end

-- ============================================================
-- SERIALIZER (now with buffer hex)
-- ============================================================
local function ser(v, d)
	d = d or 0; if d > 3 then return "<deep>" end
	local t = typeof(v)
	if t == "Instance" then
		local cls, fn = "?", "?"
		pcall(function() cls = v.ClassName end)
		pcall(function() fn = v:GetFullName() end)
		return ("<%s:%s>"):format(cls, fn)
	elseif t == "Vector3" then return ("V3(%.1f,%.1f,%.1f)"):format(v.X, v.Y, v.Z)
	elseif t == "CFrame" then local p = v.Position; return ("CF(%.1f,%.1f,%.1f)"):format(p.X, p.Y, p.Z)
	elseif t == "buffer" then
		local hex, len = bufferToHex(v, BUF_BYTES)
		return ("<buffer len=%d hex=[%s]>"):format(len, hex)
	elseif t == "EnumItem" then return tostring(v)
	elseif t == "string" then
		-- Show printable, plus hex
		local printable = v:gsub("[%c\128-\255]", ".")
		return ("\"%s\"(hex=%s,len=%d)"):format(printable, bytesToHex(v, 16), #v)
	elseif t == "number" or t == "boolean" or t == "nil" then return tostring(v)
	elseif t == "table" then
		local p, n = {}, 0
		for k, vv in pairs(v) do
			n = n + 1; if n > 12 then table.insert(p, "...") break end
			table.insert(p, ("[%s]=%s"):format(ser(k, d+1), ser(vv, d+1)))
		end
		return "{" .. table.concat(p, ", ") .. "}"
	end
	return ("<%s>"):format(t)
end
local function serArgs(...)
	local n, p = select("#", ...), {}
	for i = 1, n do p[i] = ser((select(i, ...))) end
	return table.concat(p, " | ")
end

local fireCount = 0
local mineBufferCount = 0
local markCount = 0

local R = ReplicatedStorage:WaitForChild("RemoteEvent")

pcall(function()
	local mt = getrawmetatable(game)
	local setRO = setreadonly or make_writeable
	local isRO  = isreadonly  or function() return true end
	local newcc = newcclosure or function(f) return f end
	if isRO(mt) then pcall(setRO, mt, false) end
	local oldNC = mt.__namecall
	mt.__namecall = newcc(function(self, ...)
		local method = getnamecallmethod and getnamecallmethod() or ""
		if (method == "FireServer" or method == "InvokeServer") and fireCount < MAX_FIRES then
			fireCount = fireCount + 1
			local argsStr = serArgs(...)
			-- Track mine count specifically
			local args = {...}
			if args[2] == "DamageBlockSmooth" then
				mineBufferCount = mineBufferCount + 1
			end
			local path = "?"
			pcall(function() path = self:GetFullName() end)
			pcall(log, ("[FIRE #%d] %s :%s(%s)"):format(
				fireCount, path, method, argsStr))
		end
		return oldNC(self, ...)
	end)
	pcall(setRO, mt, true)
end)

log("[Recon v3] Hook installed.")
