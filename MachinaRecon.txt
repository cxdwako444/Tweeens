-- ============================================================
-- ENV / LOG
-- ============================================================
local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end
local _writefile = find("writefile")

local LOG = {}
local function tstamp() return os.date("%H:%M:%S") end
local function log(msg)
	table.insert(LOG, ("[%s] %s"):format(tstamp(), tostring(msg)))
	if #LOG > 250 then table.remove(LOG, 1) end
end

-- ============================================================
-- REMOTE / BUFFER
-- ============================================================
local function snap(c, origin) return math.floor((c - origin) / BLOCK_SIZE + 0.5) * BLOCK_SIZE + origin end
local function makeMineBuffer(wx, wy, wz, damage)
	local b = buffer.create(32)
	buffer.writef64(b, 0,  (snap(wz, ORIGIN.Z) - ORIGIN.Z) / BLOCK_SIZE)
	buffer.writef64(b, 8,  (snap(wy, ORIGIN.Y) - ORIGIN.Y) / BLOCK_SIZE)
	buffer.writef64(b, 16, (snap(wx, ORIGIN.X) - ORIGIN.X) / BLOCK_SIZE)
	buffer.writef64(b, 24, damage)
	return b
end
local function randomNonce()
	return string.char(math.random(0,255), math.random(0,255), math.random(0,255), math.random(0,255))
end
local function fireMine(x, y, z, dmg)
	pcall(function() RE:FireServer(randomNonce(), "DamageBlockSmooth", makeMineBuffer(x, y, z, dmg)) end)
end
local function setMineMode(on)
	pcall(function() RE:FireServer(MAGIC_NONCE, "SetMode", on and "Mine" or "None") end)
end

-- ============================================================
-- TARGETS
-- ============================================================
local function collectTargets(playerPos, radius)
	local r2 = radius * radius
	local seen, result = {}, {}
	local function tryAdd(part)
		if not part:IsA("BasePart") then return end
		local p = part.Position
		local dp = p - playerPos
		local d2 = dp.X*dp.X + dp.Y*dp.Y + dp.Z*dp.Z
		if d2 > r2 then return end
		local kx = math.floor((p.X - ORIGIN.X) / BLOCK_SIZE + 0.5)
		local ky = math.floor((p.Y - ORIGIN.Y) / BLOCK_SIZE + 0.5)
		local kz = math.floor((p.Z - ORIGIN.Z) / BLOCK_SIZE + 0.5)
		local key = kx .. "," .. ky .. "," .. kz
		if seen[key] then return end
		seen[key] = true
		table.insert(result, {x=p.X, y=p.Y, z=p.Z, d2=d2})
	end
	local cp = Workspace:FindFirstChild("CollisionParts")
	if cp then for _, d in ipairs(cp:GetChildren()) do tryAdd(d) end end
	local bl = Workspace:FindFirstChild("Blocks")
	if bl then
		for _, child in ipairs(bl:GetChildren()) do
			if child:IsA("BasePart") then tryAdd(child)
			elseif child:IsA("Folder") or child:IsA("Model") then
				for _, gc in ipairs(child:GetChildren()) do tryAdd(gc) end
			end
		end
	end
	table.sort(result, function(a, b) return a.d2 < b.d2 end)
	return result
end

local stats = {
	radiusOn = false,
	pulseMode = true,        -- auto burst/rest cycle
	autoWalk = false,        -- micro-teleport for fresh chunks
	burstSec = 5,
	restSec = 2,
	radius = 24,
	maxTgts = 10,
	rateHz = 25,
	damage = 0.1,
	totalFires = 0,
	firesThisSec = 0,
	firesPerSec = 0,
	minedThisSec = 0,
	minedPerSec = 0,
	totalMined = 0,
	targetsFound = 0,
	rrIdx = 1,
	pulsePhase = "rest",
	bestMinedPerSec = 0,
}

-- ============================================================
-- MINE COUNTER
-- ============================================================
local function attachWatcher(folder)
	if not folder then return end
	folder.DescendantRemoving:Connect(function(c)
		if not c:IsA("BasePart") then return end
		local char = LocalPlayer.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if not hrp then return end
		local dp = c.Position - hrp.Position
		if dp.X*dp.X + dp.Y*dp.Y + dp.Z*dp.Z <= MINE_PROXIMITY*MINE_PROXIMITY then
			stats.minedThisSec = stats.minedThisSec + 1
			stats.totalMined = stats.totalMined + 1
		end
	end)
end
attachWatcher(Workspace:FindFirstChild("CollisionParts"))
attachWatcher(Workspace:FindFirstChild("Blocks"))
Workspace.ChildAdded:Connect(function(c)
	if c.Name == "CollisionParts" or c.Name == "Blocks" then attachWatcher(c) end
end)

-- Per-second tick
task.spawn(function()
	while true do
		task.wait(1.0)
		stats.minedPerSec = stats.minedThisSec; stats.minedThisSec = 0
		stats.firesPerSec = stats.firesThisSec; stats.firesThisSec = 0
		if stats.minedPerSec > stats.bestMinedPerSec then
			stats.bestMinedPerSec = stats.minedPerSec
		end
		if stats.radiusOn then
			local eff = stats.firesPerSec > 0 and (stats.minedPerSec / stats.firesPerSec) or 0
			log(("%s tgts=%d f/s=%d MINED=%d  AOE=%.1fx  tot=%d"):format(
				stats.pulsePhase=="burst" and "▶" or "·",
				stats.targetsFound, stats.firesPerSec, stats.minedPerSec, eff, stats.totalMined))
		end
	end
end)

-- ============================================================
-- PULSE LOOP — alternates SetMode Mine/None
-- ============================================================
task.spawn(function()
	while true do
		if stats.radiusOn and stats.pulseMode then
			stats.pulsePhase = "burst"
			setMineMode(true)
			task.wait(stats.burstSec)
			stats.pulsePhase = "rest"
			setMineMode(false)
			task.wait(stats.restSec)
		elseif stats.radiusOn then
			stats.pulsePhase = "burst"  -- always burst when pulse off
			task.wait(0.3)
		else
			stats.pulsePhase = "rest"
			task.wait(0.2)
		end
	end
end)

-- ============================================================
-- AUTO-WALK — small CFrame nudge to find fresh clusters
-- ============================================================
task.spawn(function()
	local angle = 0
	while true do
		task.wait(2.0)
		if stats.radiusOn and stats.autoWalk then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				angle = angle + math.pi / 3
				local dx = math.cos(angle) * 6
				local dz = math.sin(angle) * 6
				pcall(function() hrp.CFrame = hrp.CFrame + Vector3.new(dx, 0, dz) end)
			end
		end
	end
end)

-- ============================================================
-- MINING LOOP — round-robin during burst
-- ============================================================
task.spawn(function()
	while true do
		if stats.radiusOn and stats.pulsePhase == "burst" then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local targets = collectTargets(hrp.Position, stats.radius)
				stats.targetsFound = #targets
				if #targets > 0 then
					local n = math.min(stats.maxTgts, #targets)
					local idx = ((stats.rrIdx - 1) % n) + 1
					stats.rrIdx = stats.rrIdx + 1
					local t = targets[idx]
					fireMine(t.x, t.y, t.z, stats.damage)
					stats.totalFires = stats.totalFires + 1
					stats.firesThisSec = stats.firesThisSec + 1
				end
			else
				stats.targetsFound = 0
			end
			task.wait(1 / math.max(1, stats.rateHz))
		else
			task.wait(0.1)
		end
	end
end)

local function pillRow(y, label, values, getter, setter)
	local lbl = Instance.new("TextLabel")
	lbl.Size=UDim2.new(0,55,0,18); lbl.Position=UDim2.new(0,5,0,y)
	lbl.BackgroundTransparency=1; lbl.Text=label
	lbl.Font=Enum.Font.GothamBold; lbl.TextSize=8; lbl.TextColor3=C.DIM
	lbl.TextXAlignment=Enum.TextXAlignment.Left; lbl.Parent=root
	local boxes = {}
	local total = 230 - 65
	for i, v in ipairs(values) do
		local pillW = (total / #values) - 3
		local b = Instance.new("TextButton")
		b.Size=UDim2.new(0,pillW,0,18)
		b.Position=UDim2.new(0, 62+(i-1)*(pillW+3), 0, y)
		b.BackgroundColor3 = (getter() == v) and C.BLUE or C.CARD
		b.AutoButtonColor=false; b.Text=tostring(v); b.Font=Enum.Font.GothamBold
		b.TextSize=8; b.TextColor3=C.TXT; b.Parent=root; corner(b,3)
		b.MouseButton1Click:Connect(function()
			setter(v)
			for j, bb in ipairs(boxes) do
				bb.BackgroundColor3 = (values[j] == v) and C.BLUE or C.CARD
			end
		end)
		boxes[i] = b
	end
end
