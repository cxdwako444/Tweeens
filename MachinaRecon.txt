--      If server reads all N, throughput multiplies by N.
--   2. DATA EVENT: try the other RemoteEvent we never used.
--      Might have its own rate limit (or none).
--   3. COMMAND VARIANTS: try "MineBlock", "BulkDamage", etc.
--      Production servers sometimes accept dev/admin commands.

-- HOW TO USE:
--   - Buttons let you switch between Single / Bulk / DataEvent
--   - Watch MINED/s after switching — if it jumps, we found a bypass
--   - "Probe Commands" button fires speculative cmd names once each

-- ============================================================
-- ENV
-- ============================================================
local function find(n)
	local v = rawget(_G, n); if v then return v end
	local ok, env = pcall(getfenv, 0); if ok and env then v = env[n]; if v then return v end end
	local ok2, gg = pcall(getgenv); if ok2 and gg then v = gg[n]; if v then return v end end
	return nil
end
local _writefile = find("writefile")

-- ============================================================
-- LOG
-- ============================================================
local LOG = {}
local function tstamp() return os.date("%H:%M:%S") end
local function log(msg)
	table.insert(LOG, ("[%s] %s"):format(tstamp(), tostring(msg)))
	if #LOG > 300 then table.remove(LOG, 1) end
end

-- ============================================================
-- REMOTE / BUFFER WRAPPERS
-- ============================================================
local function snapCoord(c, origin)
	return math.floor((c - origin) / BLOCK_SIZE + 0.5) * BLOCK_SIZE + origin
end

local function writeBlockToBuffer(b, off, wx, wy, wz, damage)
	buffer.writef64(b, off + 0,  (snapCoord(wz, ORIGIN.Z) - ORIGIN.Z) / BLOCK_SIZE)
	buffer.writef64(b, off + 8,  (snapCoord(wy, ORIGIN.Y) - ORIGIN.Y) / BLOCK_SIZE)
	buffer.writef64(b, off + 16, (snapCoord(wx, ORIGIN.X) - ORIGIN.X) / BLOCK_SIZE)
	buffer.writef64(b, off + 24, damage)
end

local function makeMineBuffer(wx, wy, wz, damage)
	local b = buffer.create(32)
	writeBlockToBuffer(b, 0, wx, wy, wz, damage)
	return b
end

local function makeBulkBuffer(positions, damage)
	local b = buffer.create(#positions * 32)
	for i, pos in ipairs(positions) do
		writeBlockToBuffer(b, (i-1) * 32, pos.x, pos.y, pos.z, damage)
	end
	return b
end

local function randomNonce()
	return string.char(
		math.random(0,255), math.random(0,255),
		math.random(0,255), math.random(0,255))
end

local function fireSingle(remote, x, y, z, dmg)
	pcall(function()
		remote:FireServer(randomNonce(), "DamageBlockSmooth",
			makeMineBuffer(x, y, z, dmg))
	end)
end

local function fireBulk(remote, positions, dmg, cmd)
	pcall(function()
		remote:FireServer(randomNonce(), cmd or "DamageBlockSmooth",
			makeBulkBuffer(positions, dmg))
	end)
end

local function setMineMode(on)
	pcall(function()
		RE:FireServer(MAGIC_NONCE, "SetMode", on and "Mine" or "None")
	end)
end

-- ============================================================
-- TARGETS
-- ============================================================
local function collectTargets(playerPos, radius)
	local r2 = radius * radius
	local seen = {}
	local result = {}
	local function tryAdd(part)
		if not part:IsA("BasePart") then return end
		local p = part.Position
		local dp = p - playerPos
		local d2 = dp.X*dp.X + dp.Y*dp.Y + dp.Z*dp.Z
		if d2 > r2 then return end
		local kx = math.floor((p.X - ORIGIN.X) / BLOCK_SIZE + 0.5)
		local ky = math.floor((p.Y - ORIGIN.Y) / BLOCK_SIZE + 0.5)
		local kz = math.floor((p.Z - ORIGIN.Z) / BLOCK_SIZE + 0.5)
		local key = kx .. "," .. ky .. "," .. kz
		if seen[key] then return end
		seen[key] = true
		table.insert(result, {x=p.X, y=p.Y, z=p.Z, d2=d2})
	end
	local cp = Workspace:FindFirstChild("CollisionParts")
	if cp then for _, d in ipairs(cp:GetChildren()) do tryAdd(d) end end
	local bl = Workspace:FindFirstChild("Blocks")
	if bl then
		for _, child in ipairs(bl:GetChildren()) do
			if child:IsA("BasePart") then tryAdd(child)
			elseif child:IsA("Folder") or child:IsA("Model") then
				for _, gc in ipairs(child:GetChildren()) do tryAdd(gc) end
			end
		end
	end
	table.sort(result, function(a, b) return a.d2 < b.d2 end)
	return result
end

local stats = {
	radiusOn = false,
	mode = "Single",          -- Single | Bulk | BulkDataEvent
	radius = 24,
	bulkSize = 10,            -- blocks per bulk call
	rateHz = 30,
	damage = 0.1,
	totalFires = 0,
	firesThisSec = 0,
	firesPerSec = 0,
	minedThisSec = 0,
	minedPerSec = 0,
	totalMined = 0,
	targetsFound = 0,
	rrIdx = 1,
}

-- ============================================================
-- MINE COUNTER
-- ============================================================
local function attachWatcher(folder)
	if not folder then return end
	folder.DescendantRemoving:Connect(function(c)
		if not c:IsA("BasePart") then return end
		local char = LocalPlayer.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if not hrp then return end
		local dp = c.Position - hrp.Position
		if dp.X*dp.X + dp.Y*dp.Y + dp.Z*dp.Z <= MINE_PROXIMITY*MINE_PROXIMITY then
			stats.minedThisSec = stats.minedThisSec + 1
			stats.totalMined = stats.totalMined + 1
		end
	end)
end
attachWatcher(Workspace:FindFirstChild("CollisionParts"))
attachWatcher(Workspace:FindFirstChild("Blocks"))
Workspace.ChildAdded:Connect(function(c)
	if c.Name == "CollisionParts" or c.Name == "Blocks" then attachWatcher(c) end
end)

-- Per-second tick
task.spawn(function()
	while true do
		task.wait(1.0)
		stats.minedPerSec = stats.minedThisSec; stats.minedThisSec = 0
		stats.firesPerSec = stats.firesThisSec; stats.firesThisSec = 0
		if stats.radiusOn then
			log(("[%s] tgts=%d f/s=%d MINED/s=%d tot=%d"):format(
				stats.mode, stats.targetsFound, stats.firesPerSec,
				stats.minedPerSec, stats.totalMined))
		end
	end
end)

-- ============================================================
-- MINING LOOP — dispatch based on mode
-- ============================================================
task.spawn(function()
	while true do
		if stats.radiusOn then
			local char = LocalPlayer.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local targets = collectTargets(hrp.Position, stats.radius)
				stats.targetsFound = #targets
				if #targets > 0 then
					if stats.mode == "Single" then
						-- Round-robin single call
						local n = math.min(10, #targets)
						local idx = ((stats.rrIdx - 1) % n) + 1
						stats.rrIdx = stats.rrIdx + 1
						local t = targets[idx]
						fireSingle(RE, t.x, t.y, t.z, stats.damage)
						stats.totalFires = stats.totalFires + 1
						stats.firesThisSec = stats.firesThisSec + 1

					elseif stats.mode == "Bulk" then
						-- Pack N blocks into one buffer, one FireServer call
						local n = math.min(stats.bulkSize, #targets)
						local pack = {}
						for i = 1, n do pack[i] = targets[i] end
						fireBulk(RE, pack, stats.damage)
						stats.totalFires = stats.totalFires + 1
						stats.firesThisSec = stats.firesThisSec + 1

					elseif stats.mode == "BulkDataEvent" then
						-- Same as Bulk but use the DataEvent remote
						if DE then
							local n = math.min(stats.bulkSize, #targets)
							local pack = {}
							for i = 1, n do pack[i] = targets[i] end
							fireBulk(DE, pack, stats.damage)
							stats.totalFires = stats.totalFires + 1
							stats.firesThisSec = stats.firesThisSec + 1
						end
					end
				end
			else
				stats.targetsFound = 0
			end
			task.wait(1 / math.max(1, stats.rateHz))
		else
			task.wait(0.1)
		end
	end
end)

-- ============================================================
-- PROBE: try speculative command names once
-- ============================================================
local function probeCommands()
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then log("probe: no character"); return end
	local targets = collectTargets(hrp.Position, 18)
	if #targets == 0 then log("probe: no targets"); return end
	local t = targets[1]
	local cmds = {
		"DamageBlock", "MineBlock", "BulkDamage", "BulkMine",
		"BreakBlock", "DestroyBlock", "AreaMine", "RadiusMine",
		"InstantMine", "AOEDamage", "ChunkDamage", "DamageChunk",
		"DamageBlocks", "DamageBlocksSmooth", "MineArea",
	}
	log("probe: firing " .. #cmds .. " speculative cmds")
	local before = stats.totalMined
	for _, cmd in ipairs(cmds) do
		pcall(function()
			RE:FireServer(randomNonce(), cmd,
				makeMineBuffer(t.x, t.y, t.z, stats.damage))
		end)
		task.wait(0.1)
	end
	task.wait(0.5)
	local diff = stats.totalMined - before
	log(("probe done: mined +%d during probe"):format(diff))
end
